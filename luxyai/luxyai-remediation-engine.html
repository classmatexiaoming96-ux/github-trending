195    """把 AI 的自然语言步骤绑定到平台真实可执行的只读探针。"""

220def expert_steps_from_diagnosis(diagnosis: dict) -> list[dict[str, Any]]:

232            expected = raw.get("expected_evidence") or raw.get("evidence") or []
233        else:
