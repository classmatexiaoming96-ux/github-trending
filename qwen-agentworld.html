  21    3. Nested tags: Handles tags nested within each other (e.g., <think>...<think>...</think>...</think>).
  22    4. Left unclosed: Removes content before an orphaned </think>.
  23    5. Right unclosed: Removes content starting from an orphaned <think>.
  24       - If `response_tag` is provided, stops removal at the first occurrence of `response_tag` 
  25         to preserve the actual output.
  26       - Otherwise, removes everything to the end of the string.
  27    
  28    Args:
  29        text: Input text to clean
  30        response_tag: Optional response tag name to use as a boundary for right-unclosed tags.
  31                      This prevents deleting the valid response if the thinking tag is unclosed.
  32        
  33    Returns:
  34        Cleaned text with all thinking content removed.
  35    """
  36    if not text:
  37        return text
  38    
  39    # Find all <think> and </think> tags
  40    tags = []
  41    # Find opening tags
  42    for m in re.finditer(r"<think>", text, re.IGNORECASE):
  43        tags.append((m.start(), "open"))
  44    # Find closing tags
  45    for m in re.finditer(r"</think>", text, re.IGNORECASE):
  46        tags.append((m.start(), "close"))
  47    
  48    if not tags:
  49        return text
  50    
  51    # Sort by position
  52    tags.sort(key=lambda x: x[0])
  53    
  54    # Build list of ranges to remove
  55    ranges_to_remove = []
  56    used_indices = set()
  57    
  58    # First pass: match opening tags with corresponding closing tags
  59    for i, (pos, tag_type) in enumerate(tags):
  60        if i in used_indices:
  61            continue
  62        if tag_type == "open":
  63            # Find the next closing tag
  64            for j in range(i + 1, len(tags)):
  65                if j in used_indices:
  66                    continue
  67                j_pos, j_type = tags[j]
  68                if j_type == "close":
  69                    # Found matching pair
  70                    close_tag_len = len("</think>")
  71                    ranges_to_remove.append((pos, j_pos + close_tag_len))
  72                    used_indices.add(i)
  73                    used_indices.add(j)
  74                    break

 137    """Parse and extract predicted output from model generation.
 138    
 139    The extraction process is designed to be robust against "thinking models" that might
 140    output fake response tags during their reasoning process.
 141    
 142    Steps:
 143    1. Remove all thinking tags (and their content) using `_remove_thinking_tags`.
 144    2. Extract content from the **LAST** valid response tag block found in the cleaned text.
 145       - Prioritizing the last block ensures we capture the final answer, ignoring any 
 146         fake or intermediate blocks that might have survived thinking removal (though unlikely).
 147    
 148    Args:
 149        raw_output: Raw model generation text
 150        response_tag: Tag name to extract content from (e.g., "predicted_observation")
 151        
 152    Returns:
 153        Extracted output string, or cleaned text if no tag found
 154    """
 155    if not raw_output:
 156        return "No output"
 157    
 158    # Step 1: Remove thinking tags
 159    cleaned_text = _remove_thinking_tags(raw_output, response_tag)
 160    
 161    # Step 2: Extract content from response tag
 162    # We want the LAST valid block.
 163    
 164    # Find all start tags
 165    start_pattern = rf"<{re.escape(response_tag)}>"
 166    start_matches = list(re.finditer(start_pattern, cleaned_text, re.IGNORECASE))
 167

 169        # Fallback: return the cleaned text as-is (model may not use XML tags)
 170        return cleaned_text.strip() if cleaned_text.strip() else "No output"
 171    
 172    # Take the last start tag
 173    last_start_match = start_matches[-1]
 174    start_pos = last_start_match.end()
 175
