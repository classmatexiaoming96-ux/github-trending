  62def _clean_for_summary(history_text):
  63    """Strip history.md's markdown chrome so the extractive summarizer sees prose,
  64    not headings / role labels / tool bullets."""
  65    out = []
  66    for line in history_text.splitlines():
  67        s = line.strip()
  68        if not s or s.startswith("#"):
  69            continue
  70        if s.startswith("- ") or s.startswith("`"):  # tool-call bullet lines
  71            continue
  72        # Drop the **You:** / **Claude:** role prefixes, keep the spoken text.
  73        for prefix in ("**You:**", "**Claude:**"):
  74            if s.startswith(prefix):
  75                s = s[len(prefix):].strip()
  76        if s:
  77            out.append(s)
  78    return "\n".join(out)

  75def _select(sentences, scores, k):

  96def _idf(tokens):
  97    n = len(tokens)
  98    df = Counter()
  99    for toks in tokens:
 100        for w in set(toks):
 101            df[w] += 1
 102    return {w: math.log((1.0 + n) / (1.0 + d)) + 1.0 for w, d in df.items()}
 103

 107def _tfidf_sparse(tokens, idf):
 108    vecs = []
 109    for toks in tokens:
 110        if not toks:
 111            vecs.append({})
 112            continue
 113        tf = Counter(toks)
 114        length = len(toks)
 115        vec = {w: (c / length) * idf[w] for w, c in tf.items()}
 116        norm = math.sqrt(sum(v * v for v in vec.values())) or 1.0
 117        vecs.append({w: v / norm for w, v in vec.items()})
 118    return vecs
 119

 121def _cosine(a, b):
 122    if len(a) > len(b):
 123        a, b = b, a
 124    return sum(val * b.get(w, 0.0) for w, val in a.items())

 127def _textrank_python(tokens, idf):
 128    vecs = _tfidf_sparse(tokens, idf)
 129    n = len(vecs)
 130    sim = [[0.0] * n for _ in range(n)]
 131    for i in range(n):
 132        for j in range(i + 1, n):
 133            c = _cosine(vecs[i], vecs[j])
 134            if c:
 135                sim[i][j] = sim[j][i] = c
 136    row_sum = [sum(row) for row in sim]
 137
 138    scores = [1.0 / n] * n
 139    base = (1.0 - _DAMPING) / n
 140    for _ in range(_MAX_ITERS):
 141        new = [base] * n
 142        for i in range(n):
 143            if row_sum[i] == 0.0:
 144                continue
 145            share = _DAMPING * scores[i] / row_sum[i]
 146            row = sim[i]
 147            for j in range(n):
 148                w = row[j]
 149                if w:
 150                    new[j] += share * w
 151        if sum(abs(new[i] - scores[i]) for i in range(n)) < _TOL:
 152            scores = new
 153            break
 154        scores = new
 155    return scores

 160def _textrank_numpy(tokens, idf):
 161    import numpy as np
 162
 163    vocab = sorted(idf)
 164    index = {w: i for i, w in enumerate(vocab)}
 165    n, vsize = len(tokens), len(vocab)
 166
 167    mat = np.zeros((n, vsize), dtype=np.float64)
 168    for i, toks in enumerate(tokens):
 169        if not toks:
 170            continue
 171        length = len(toks)
 172        for w, c in Counter(toks).items():
 173            mat[i, index[w]] = (c / length) * idf[w]
 174
 175    norms = np.sqrt((mat * mat).sum(axis=1, keepdims=True))
 176    norms[norms == 0] = 1.0
 177    mat /= norms
 178
 179    sim = mat @ mat.T
 180    np.fill_diagonal(sim, 0.0)
 181    row_sum = sim.sum(axis=1, keepdims=True)
 182    row_sum[row_sum == 0] = 1.0
 183    transition = sim / row_sum
 184
 185    base = (1.0 - _DAMPING) / n
 186    scores = np.full(n, 1.0 / n)
 187    for _ in range(_MAX_ITERS):
 188        prev = scores
 189        scores = base + _DAMPING * (transition.T @ scores)
 190        if np.abs(scores - prev).sum() < _TOL:
 191            break
 192    return scores.tolist()
