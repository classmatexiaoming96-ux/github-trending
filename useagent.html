delta accumulator 兼顾直播和持久化packages/agent-harness/src/bridge.ts:129
129export class NativeBridgeDeltaAccumulator {
130  #messageText = new Map<string, DurableTextSegment>();
131  #reasoningText = new Map<string, DurableTextSegment>();
132  #encoder = new TextEncoder();
133
134  #key(body: { readonly messageId: string; readonly ownerChildId?: string }): string {
135    return `${body.ownerChildId ?? "root"}\0${body.messageId}`;
136  }
137
138  #append(
139    body: Extract<NativeBridgeFrameBody, { kind: "message.delta" | "reasoning.delta" }>,
140    target: Map<string, DurableTextSegment>,
141  ): readonly NativeBridgeFrameBody[] {
142    const key = this.#key(body);
143    let segment = target.get(key) ?? { index: 0, text: "", bytes: 0 };
144    const output: NativeBridgeFrameBody[] = [];
145    let changed = false;
146    for (const character of body.text) {
147      const bytes = this.#encoder.encode(character).byteLength;
148      if (segment.bytes > 0 && segment.bytes + bytes > NATIVE_BRIDGE_DURABLE_TEXT_BYTES) {
149        if (changed) output.push({ ...body, text: segment.text, segment: segment.index });
150        segment = { index: segment.index + 1, text: "", bytes: 0 };
151        changed = false;
152      }
153      segment.text += character;
154      segment.bytes += bytes;
155      changed = true;
156    }
157    if (changed) output.push({ ...body, text: segment.text, segment: segment.index });
158    target.set(key, segment);
159    return output;
160  }
161
162  #replace(
163    body: Extract<NativeBridgeFrameBody, { kind: "message.authoritative" | "reasoning.authoritative" }>,
164    target: Map<string, DurableTextSegment>,
165  ): readonly NativeBridgeFrameBody[] {
166    const key = this.#key(body);
167    const previousLastIndex = target.get(key)?.index ?? -1;
168    target.delete(key);
169    const kind = body.kind === "message.authoritative" ? "message.delta" : "reasoning.delta";
170    const replacement = this.#append(
171      {
172        kind,
173        messageId: body.messageId,
174        text: body.text,
175        authoritative: true,
176        ...(body.ownerChildId ? { ownerChildId: body.ownerChildId } : {}),
177      },
178      target,
179    );
180    const authoritativeReplacement = replacement.length > 0
181      ? replacement
182      : [{
183          kind,
184          messageId: body.messageId,
185          text: "",
186          segment: 0,
187          authoritative: true,
188          ...(body.ownerChildId ? { ownerChildId: body.ownerChildId } : {}),
189        } as const];
190    const replacementLastIndex = target.get(key)?.index ?? -1;
191    const tombstones: NativeBridgeFrameBody[] = [];
192    for (let index = replacementLastIndex + 1; index <= previousLastIndex; index++) {
193      tombstones.push({
194        kind,
195        messageId: body.messageId,
196        text: "",
197        segment: index,
198        authoritative: true,
199        ...(body.ownerChildId ? { ownerChildId: body.ownerChildId } : {}),
200      });
201    }
202    return [...authoritativeReplacement, ...tombstones];
203  }
204
205  durable(body: NativeBridgeFrameBody): readonly NativeBridgeFrameBody[] {
206    if (body.kind === "message.delta") return this.#append(body, this.#messageText);
207    if (body.kind === "reasoning.delta") return this.#append(body, this.#reasoningText);
208    if (body.kind === "message.authoritative") return this.#replace(body, this.#messageText);
209    if (body.kind === "reasoning.authoritative") return this.#replace(body, this.#reasoningText);
210    return [body];
211  }
212}
