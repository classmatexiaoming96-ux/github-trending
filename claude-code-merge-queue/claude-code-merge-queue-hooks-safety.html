76export function createLane(mainTop: string, cfg: ClaudeCodeMergeQueueConfig): { wt: string; branch: string; lane: number } {
77  // Clean up administrative entries for worktrees whose directories are
78  // already gone (e.g. someone `rm -rf`'d one instead of `git worktree
79  // remove`) so reusing that lane number's branch name doesn't fail with
80  // "already checked out elsewhere."
81  tryGit(["worktree", "prune"], mainTop);
82
83  const repoName = basename(mainTop);
84  let lane = 0;
85  for (;;) {
86    lane += 1;
87    if (lane > MAX_LANE_ATTEMPTS) {
88      throw new Error(`could not claim a lane after ${MAX_LANE_ATTEMPTS} attempts — is mainTop ('${mainTop}') actually the repo root?`);
89    }
90    const wt = join(dirname(mainTop), `${repoName}${cfg.worktreeSuffix}${lane}`);
91    if (existsSync(wt)) continue; // still claimed — try the next lane
92
93    const branch = `${cfg.branchPrefix}${lane}`;
94    const branchExists = tryGit(["show-ref", "--verify", "--quiet", `refs/heads/${branch}`], mainTop).ok;
95
96    // Base every new lane on the main checkout's own current HEAD — never on
97    // origin/integrationBranch. The main checkout is kept fast-forwarded by
98    // `sync` precisely so its local HEAD IS the trusted, up-to-date view;
99    // reading origin directly instead can be *behind* local HEAD (a commit
100    // made here but not yet pushed) and silently drop it from every new
101    // lane. That's not a hypothetical: it breaks the literal Quickstart —
102    // `init` writes claude-code-merge-queue.config.mjs/CLAUDE.md/.claude locally, and the
103    // very first lane created before that commit is pushed anywhere would
104    // otherwise come up with none of it.
105    let add;
106    if (branchExists) {
107      add = tryGit(["worktree", "add", wt, branch], mainTop);
108    } else {
109      add = tryGit(["worktree", "add", wt, "-b", branch], mainTop);
110    }
111
112    if (!add.ok) {
113      // Someone else claimed this exact lane between our existsSync check
114      // and `git worktree add` — the same race a `mkdir` guards against.
115      // Try the next one.
116      continue;
117    }
118
119    for (const rel of cfg.symlinks) {
120      const src = join(mainTop, rel);
121      const dest = join(wt, rel);
122      if (!existsSync(src) || existsSync(dest)) continue;
123      try {
124        mkdirSync(dirname(dest), { recursive: true });
125        symlinkSync(src, dest);
126      } catch {
127        /* best-effort — a missing symlink degrades to "run npm install," not a hard failure */
128      }
129    }
130
131    return { wt, branch, lane };
132  }

157// nowhere to install claude-code-merge-queue INTO; npx's ephemeral cache is the only way

121export type HooksPathResult = "set" | "already-set" | "custom-path";
122
123/**
124 * A `.husky/pre-push` file on disk enforces nothing on its own — git only
125 * runs it if `core.hooksPath` points somewhere that resolves to it, which is
126 * normally set as a side effect of the package manager's install step
127 * (husky's own `prepare` script). On a freshly cloned repo where nobody's
128 * run that install yet — the exact state Quickstart leaves you in right
129 * after `init` — the file is silently inert and a direct push sails through
130 * uncontested. Since Claude Code Merge Queue is the one promising "pushes are gated now,"
131 * it sets this itself instead of depending on a step that may not have
132 * happened yet, mirroring exactly what `husky install` itself does.
133 *
134 * Husky v9 changed its own convention mid-flight: v6–v8 point
135 * core.hooksPath directly at `.husky` (hook files run as-is); v9 points it
136 * at `.husky/_`, a generated wrapper directory that then execs the real
137 * `.husky/<hookname>` file. Both are legitimate, already-correct setups —
138 * only treat something OTHER than either as a deliberate custom path worth
139 * warning about. `.husky/_` doesn't exist yet on the fresh-clone/no-install
140 * case this function exists for, so `.husky` remains the right thing to set
141 * when nothing's configured at all; if the project turns out to be v9,
142 * husky's own next real install corrects it to `.husky/_`.
143 */
144export function ensureHooksPath(root: string): HooksPathResult {
145  let current: string | null;
146  try {
147    current = execFileSync("git", ["config", "core.hooksPath"], { cwd: root, encoding: "utf8" }).trim();
148  } catch {
149    current = null; // unset
150  }
151
152  if (current === ".husky" || current === ".husky/_") return "already-set";
153  if (current) return "custom-path"; // respect an existing deliberate setup — don't override it
154
155  execFileSync("git", ["config", "core.hooksPath", ".husky"], { cwd: root });
156  return "set";
157}

44/**
45 * Run the configured check command. Returns the exit code to propagate.
46 *
47 * A null checkCommand is only ever silent if checksRequired was explicitly
48 * turned off — the default is to FAIL the push, because a merge queue that
49 * lets unverified code through by default isn't one, it's a false sense of
50 * safety.
51 *
52 * Always runs from `root`, not whatever the caller's cwd happens to be. In
53 * the real git-push flow that's moot — git always resets cwd to the repo
54 * root before running a hook — but `check-push` is also a directly
55 * user-runnable command, and silently depending on git's hook behavior for
56 * correctness (instead of just being correct on its own) is exactly the
57 * kind of implicit assumption that bites the first time someone runs it by
58 * hand from a subdirectory.
59 */
60export function runCheckCommand(cfg: Pick<ClaudeCodeMergeQueueConfig, "checkCommand" | "checksRequired">, root: string): number {
61  if (!cfg.checkCommand) {
62    if (cfg.checksRequired) {
63      console.error([
64        "",

66        "   This push would land with NOTHING verifying it — no lint, no test, no build.",
67        "   Set checkCommand in claude-code-merge-queue.config.mjs, e.g. \"npm run check\".",
68        "   Or, if you really have nothing to check yet, set checksRequired: false —",
69        "   deliberately, so it's a visible, committed choice, not a silent gap.",
70        "",
71      ].join("\n"));
72      return 1;
73    }
74    console.log("claude-code-merge-queue check-push: no checkCommand configured (checksRequired: false — running with no checks, on purpose).");
75    return 0;
76  }
77
78  console.log(`claude-code-merge-queue check-push: running "${cfg.checkCommand}"…`);

80  if (result.status !== 0) {
