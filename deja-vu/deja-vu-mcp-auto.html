27func serveMCP(r io.Reader, w io.Writer) error {
28	s := bufio.NewScanner(r)
29	// MCP messages are line-delimited JSON here. Allow large, but bounded, client lines.
30	s.Buffer(make([]byte, 0, 64*1024), 10*1024*1024)
31	enc := json.NewEncoder(w)
32	for s.Scan() {
33		line := strings.TrimSpace(s.Text())
34		if line == "" {
35			continue
36		}
37		var req rpcRequest
38		if err := json.Unmarshal([]byte(line), &req); err != nil {
39			writeRPCError(enc, nil, -32700, "parse error")
40			continue
41		}
42		if req.ID == nil {
43			// Notification. Do not reply.
44			continue
45		}
46		result, code, msg := handleMCP(req)
47		if code != 0 {
48			writeRPCError(enc, req.ID, code, msg)
49			continue
50		}
51		if err := enc.Encode(map[string]any{"jsonrpc": "2.0", "id": req.ID, "result": result}); err != nil {
52			return err
53		}
54	}
55	if err := s.Err(); err != nil {
56		// Oversized or malformed client frames are reported as JSON-RPC parse errors,
57		// then the stdio server exits gracefully instead of silently hard-stopping.
58		writeRPCError(enc, nil, -32700, "parse error")
59		if os.Getenv("DEJA_DEBUG") == "1" {
60			fmt.Fprintf(os.Stderr, "deja mcp scanner error: %v\n", err)
61		}
62	}
63	return nil
64}

152func recallText(q, harness string, limit, budget int) (string, error) {
153	if limit <= 0 {
154		limit = 5
155	}
156	o := search.Options{Query: q, Harness: harness, All: true}
157	if err := index.EnsureForSearch(index.DefaultDir(), o, false, mcpProgress()); err != nil {
158		return "", err
159	}
160	ss, err := index.SearchWithRecovery(index.DefaultDir(), o, mcpProgress())
161	if err != nil {
162		return "", err
163	}
164	hits, err := search.Run(ss, o)
165	if err != nil {
166		return "", err
167	}
168	if len(hits) == 0 {
169		return fmt.Sprintf("No prior deja sessions matched %q.", q), nil
170	}
171	if len(hits) > limit {
172		hits = hits[:limit]
173	}
174	var b strings.Builder
175	fmt.Fprintf(&b, "deja recall for %q (%d match(es))\n", q, len(hits))
176	for i, h := range hits {
177		fmt.Fprintf(&b, "\n%d. [%s] %s · %s · %d matches", i+1, h.Session.Harness, h.Session.Project, h.Session.ID, h.Count)
178		if !h.Session.Updated.IsZero() {
179			fmt.Fprintf(&b, " · updated %s", h.Session.Updated.Format("2006-01-02"))
180		}
181		fmt.Fprintln(&b)
182		for _, sn := range h.Snippets {
183			fmt.Fprintf(&b, "- %s\n", sn)
184		}
185		if b.Len() >= budget {
186			break
187		}
188	}
189	out := b.String()
190	if len(out) > budget {
191		out = trimUTF8(out, budget)
192	}
193	return out, nil
194}
