▸压缩不是删除历史，而是替换为摘要

src/agent.ts:37 先检查 abort 和长度，src/agent.ts:41 切出旧消息，src/agent.ts:50 构造摘要 context，src/agent.ts:57 用同一个 stream 总结。失败时 src/agent.ts:63 直接保留原始 context，这是避免空摘要污染会话的关键。

37async function compactContext(model: Model, context: Context, signal?: AbortSignal): Promise<void> {
38  if (signal?.aborted) return  // abort 时不压缩——避免空摘要损坏 context
39  if (context.messages.length < COMPACT_THRESHOLD) return
40
41  const oldMessages = context.messages.slice(0, -KEEP_RECENT)
42  const recentMessages = context.messages.slice(-KEEP_RECENT)
43
44  // 把旧消息序列化成纯文本让 LLM 总结
45  const conversationText = oldMessages
46    .map(m => `${m.role}: ${typeof m.content === 'string' ? m.content : JSON.stringify(m.content)}`)
47    .join('\n')
48
49  // 用一轮非流式调用让 LLM 总结
50  const summaryContext: Context = {
51    systemPrompt: '请将以下对话总结为简洁的上下文摘要，保留关键决策、已做的工作和待办事项。',
52    messages: [{ role: 'user', content: conversationText }],
53  }
54
55  let summary = ''
56  let failed = false
57  for await (const ev of stream(model, summaryContext, { signal })) {
58    if (ev.type === 'text_delta') summary += ev.delta
59    else if (ev.type === 'error' || ev.type === 'done' && ev.stopReason === 'aborted') { failed = true; break }
60  }
61
62  // 压缩失败时不替换 context——保留原始消息比空摘要更安全
63  if (failed || !summary) return
64
65  // 用摘要消息替换旧消息，保留近期消息
66  context.messages = [
67    { role: 'user', content: `[context summary]\n${summary}` },
68    ...recentMessages,
69  ]
70}
