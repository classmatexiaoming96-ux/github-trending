100        if (prior && prior !== choice.harnessId) {
101          await adapters.get(prior)?.turns.resetSession?.(input.session.id);
102          await adapter.turns.resetSession?.(input.session.id);
103        }
104        lastHarness.set(input.session.id, choice.harnessId);
105        return adapter.turns.runTurn({ ...input, harness: choice.harnessId, model: choice.modelId });
106      },
107      async resetSession(sessionId) {
108        lastHarness.delete(sessionId);
109        await Promise.all([...adapters.values()].map((adapter) => adapter.turns.resetSession?.(sessionId)));
110      },
111      async close() {
112        await Promise.all([...new Set(adapters.values())].map((adapter) => adapter.turns.close?.()));
113      },
114    },
115  };

23  const token = run.leaseToken;
24  if (token === null) throw new Error(`processRun called with an unleased run ${run.id}`);
25  const intervalMs = deps.heartbeatIntervalMs ?? Math.max(1_000, Math.floor(deps.leaseTtlMs / 3));
26  const cancel = new AbortController();
27  let consecutiveLost = 0;
28  let leaseLost = false;
29  const beat = setInterval(() => {
30    void deps.runs
31      .heartbeat(run.id, token, deps.leaseTtlMs)
32      .then((alive) => {
33        if (alive) {
34          consecutiveLost = 0;
35          return;
36        }
37        consecutiveLost += 1;
38        if (consecutiveLost >= LEASE_LOST_CONSECUTIVE && !leaseLost) {
39          leaseLost = true;
40          clearInterval(beat);
41          console.warn(
42            `[worker] run ${run.id} lost its lease after ${consecutiveLost} consecutive beats; cancelling the in-process turn`,
43          );
44          cancel.abort();
45        }
46      })
47      .catch((err: unknown) => {
48        consecutiveLost = 0;
49        console.warn(`[worker] heartbeat failed for run ${run.id} (transient, ignored): ${errMessage(err)}`);
50      });
51  }, intervalMs);
52  let beatStopped = false;
53  const stopBeat = (): void => {
54    if (beatStopped) return;
55    beatStopped = true;
56    clearInterval(beat);
57  };
58  try {
59    const queueMs = run.startedAt !== null ? Math.max(0, run.startedAt - run.createdAt) : undefined;
60    const result = await deps.orchestrator.handleTurn({
61      ...run.request,

63      runId: run.id,
64      attempt: run.attempts,
65      finalAttempt: errorParks(run, deps.runs.maxClaims),
66      background: opts?.background ?? false,
67      cancel: cancel.signal,
68      ...(queueMs !== undefined ? { queueMs } : {}),
69    });
70    stopBeat();
71    await deps.runs.complete(run.id, token, result);
72    return result;
73  } catch (err) {

75    await deps.runs.fail(run.id, token, errMessage(err), {

79  } finally {

101export function createWorker(deps: WorkerDeps): Worker {
102  const workerId = deps.workerId ?? `w-${randomUUID().slice(0, 8)}`;
103  const pollMs = deps.pollMs ?? 50;
104  let stopped = false;
105  let loopDone: Promise<void> | null = null;
106  let inFlight: { runId: string; leaseToken: string; threadRef: string } | null = null;
107  let releasedLeaseToken: string | null = null;
108  let releasing: Promise<void> | null = null;
109
110  async function loop(): Promise<void> {
111    let claimFailures = 0;
112    while (!stopped) {
113      if (deps.canClaim && !deps.canClaim()) {
114        await sleep(pollMs);
115        continue;
116      }
117      let run: Run | null;
118      try {
119        run = await deps.runs.claim(workerId, deps.leaseTtlMs);
120        claimFailures = 0;
121      } catch (e) {
122        claimFailures += 1;
123        if (claimFailures >= CLAIM_FAIL_CRASH_CONSECUTIVE) throw e;
124        swallow("worker: claim failed (transient, retrying)", e);
125        await sleep(Math.min(pollMs * 2 ** Math.min(claimFailures, 5), 5_000));
126        continue;
127      }
128      if (!run) {
129        await sleep(pollMs);
130        continue;
131      }
132      if (stopped) {
133        if (run.leaseToken !== null)
134          await deps.runs
135            .releaseLease(run.id, run.leaseToken)
136            .catch((e) => swallow("worker: post-stop claim handback failed", e));
137        break;
138      }
139      inFlight =
140        run.leaseToken !== null ? { runId: run.id, leaseToken: run.leaseToken, threadRef: run.sessionId } : null;
141      deps.onClaimed?.();
142      try {
143        await processRun(deps, run, { background: true });
144      } catch (e) {
145        swallow("worker: background run crashed", e);
146      } finally {
147        inFlight = null;
148      }
149    }
150  }
151
152  return {
153    start() {
154      if (loopDone) return;
155      stopped = false;
156      loopDone = loop();
157    },
158    busy() {
159      return inFlight !== null;
160    },
161    releaseInFlight() {
162      const held = inFlight;
163      if (!held || held.leaseToken === releasedLeaseToken) return Promise.resolve();
164      if (releasing) return releasing;
165      releasing = (async () => {
166        try {
167          if (await deps.runs.heartbeat(held.runId, held.leaseToken, deps.leaseTtlMs)) {
168            const session = await deps.sessions.getByThread(held.threadRef);
169            if (session) await deps.sessions.forceReleaseLease(session.id);
170            await deps.runs.releaseLease(held.runId, held.leaseToken);
171          }
172          releasedLeaseToken = held.leaseToken;
173        } catch (e) {
174          swallow("worker: releaseInFlight failed", e);
175        } finally {
176          releasing = null;
177        }
178      })();
179      return releasing;
180    },
181    async stop(drainMs = STOP_DRAIN_MS) {
182      stopped = true;
183      await Promise.race([loopDone, sleep(drainMs, { unref: true })]);
184      loopDone = null;
185    },
186  };
187}
