128def _openclaw_is_relay_mode(args: argparse.Namespace) -> bool:
129    """OpenClaw browser-driver mode dispatch by image tag.
130
131    The explicitly named legacy images (`commercecraftbench/openclaw:base-v1.3`
132    and `commercecraftbench/openclaw:ccb-v1.3`) are built on an OpenClaw
133    2026.3.x runtime base. That release ships a Chrome extension `Browser
134    Relay` that we patch + attach via CDP.
135    Profile name = `chrome-relay`, driver = `extension`, cdpUrl points
136    at an in-process relay server.
137
138    v2026.5.22 dropped the extension entirely (`assets/chrome-extension/`
139    removed, no `chrome-relay` profile in dist) and only supports two
140    drivers — `openclaw` (managed) and `existing-session` (limited). The
141    correct config for the upstream-entrypoint Chrome at 9222 is:
142        driver=openclaw, attachOnly=true, cdpUrl=http://127.0.0.1:9222
143    which yields `local-managed` capability (full snapshot+act+tabs) while
144    re-using the noVNC-visible Chrome from the entrypoint. We previously
145    used driver=existing-session, but that caps capabilities to
146    `local-existing-session` (no selector targeting, no managed tabs) and
147    triggers the "user-style profile cannot use sandbox browser" error.
148
149    Detection is deliberately opt-in for those two legacy tags. The public
150    release pins the May image by digest, so looking for ``2026.5.22`` in the
151    image string would misclassify the canonical digest as the March runtime.
152    Current and custom images therefore use managed-attach mode by default.
153    """
154    image = (getattr(args, "image", "") or "").lower()
155    legacy_tags = (
156        "commercecraftbench/openclaw:base-v1.3",
157        "commercecraftbench/openclaw:ccb-v1.3",
158    )
159    return any(tag in image for tag in legacy_tags)

162def collect_openclaw_state_artifacts(container: str, agent_dir: Path) -> dict[str, Any]:
163    """Tar /root/.openclaw/agents/main/ + copy /tmp/openclaw/openclaw-*.log
164    out of the container into ``agent_dir`` for post-mortem inspection.
165
166    Mirrors the spirit of ``collect_accio_state_artifacts`` but the OpenClaw
167    container is destroyed per-task, so there is no host-side cleanup pass.
168    Returns a dict for the manifest / debug file.
169    """
170    out: dict[str, Any] = {"collected": False, "files": [], "openclaw_agents_present": False}
171
172    # 1. agents/main tarball — quick check first to avoid a noisy tar of a
173    # missing path.
174    probe = docker(
175        "exec", container, "bash", "-lc",
176        "test -d /root/.openclaw/agents/main && echo yes || echo no",
177        check=False, capture=True,
178    )
179    has_agents = (probe.stdout or "").strip() == "yes"
180    out["openclaw_agents_present"] = has_agents
181    if has_agents:
182        tar_path = agent_dir / "openclaw_state.tar.gz"
183        try:
184            with tar_path.open("wb") as tf:
185                proc = subprocess.run(
186                    [
187                        "docker", "exec", container, "bash", "-lc",
188                        "tar --exclude=main/agent/auth-profiles.json "
189                        "--exclude=main/agent/credentials.json "
190                        "-czf - -C /root/.openclaw/agents main",
191                    ],
192                    stdout=tf,
193                    stderr=subprocess.PIPE,
194                    check=False,
195                )
196            if proc.returncode == 0 and tar_path.is_file() and tar_path.stat().st_size > 0:
197                out["files"].append(str(tar_path.relative_to(agent_dir)))
198                # Untar locally for grep-ability.
199                state_dir = agent_dir / "openclaw_state"
200                state_dir.mkdir(parents=True, exist_ok=True)
201                untar = subprocess.run(
202                    ["tar", "-xzf", str(tar_path), "-C", str(state_dir)],
203                    capture_output=True, text=True, check=False,
204                )
205                if untar.returncode == 0:
206                    out["files"].append(str(state_dir.relative_to(agent_dir)) + "/")
207                else:
208                    out["untar_error"] = (untar.stderr or "").strip()[:500]
209            else:
210                out["tar_error"] = (proc.stderr.decode("utf-8", errors="replace") if isinstance(proc.stderr, bytes) else (proc.stderr or "")).strip()[:500]
211                if tar_path.is_file() and tar_path.stat().st_size == 0:
212                    tar_path.unlink(missing_ok=True)
213        except OSError as exc:
214            out["tar_error"] = str(exc)
215
216    # 2. OpenClaw runtime logs at /tmp/openclaw/openclaw-*.log. List first,
217    # then docker cp each. Tolerate misses.
218    list_proc = docker(
219        "exec", container, "bash", "-lc",
220        "ls -1 /tmp/openclaw/openclaw-*.log 2>/dev/null || true",
221        check=False, capture=True,
222    )
223    log_paths = [line.strip() for line in (list_proc.stdout or "").splitlines() if line.strip()]
224    if log_paths:
225        runtime_dir = agent_dir / "openclaw_runtime_logs"
226        runtime_dir.mkdir(parents=True, exist_ok=True)
227        for remote_log in log_paths:
228            local_dst = runtime_dir / Path(remote_log).name
229            cp = docker("cp", f"{container}:{remote_log}", str(local_dst), check=False, capture=True)
230            if cp.returncode == 0 and local_dst.is_file():
231                out["files"].append(str(local_dst.relative_to(agent_dir)))
232
233    if out["files"]:
234        out["collected"] = True
235    elif not has_agents:
236        out["reason"] = "openclaw_agents_dir_missing"

240def _parse_openclaw_provider_headers(raw_headers: list[str]) -> dict[str, str]:
241    """Parse repeated `--openclaw-provider-header KEY:VALUE` flags into a dict.
242
243    Only the FIRST colon is treated as the separator, so values that
244    themselves contain ``:`` (URLs, times) round-trip verbatim.
245    """
246    parsed: dict[str, str] = {}
247    for raw in raw_headers or []:
248        if ":" not in raw:
249            raise SystemExit(
250                f"--openclaw-provider-header must be formatted 'Key:Value' (got: {raw!r})"
251            )
252        key, _, value = raw.partition(":")
253        key = key.strip()
254        value = value.strip()
255        if not key:
256            raise SystemExit(
257                f"--openclaw-provider-header has empty key (got: {raw!r})"
258            )
259        parsed[key] = value
260    return parsed

§ 02 · KEY FLOW

关键流程

task config → contract object → runtime materialization → agent execution → verifier/report

从这些源码行可以看到，RealReplicaBench 的流程不是把 Agent 输出丢给一个脚本评分，而是先构造可复现任务世界，再让 harness 运行，最后把 verifier、trajectory 和 report 合并成证据链。

边界 / 失败模式

▸失败在哪里会被放大

  • 如果任务声明不完整，src/real_replica_bench/harnesses/openclaw/runner.py:78 附近的解析会提前中断，而不是让 Agent 在缺环境时跑出伪结果。

  • 如果 Agent 只完成部分业务动作，最终分数会被 verifier 端硬门禁压成失败，需要回看 trajectory。
