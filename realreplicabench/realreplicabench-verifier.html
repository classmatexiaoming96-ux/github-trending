119    reward: dict[str, Any],
120    *,
121    verifier_exit: int,
122    agent_exec_returncode: int | None,
123    agent_early_terminated: bool = False,
124) -> dict[str, Any]:
125    """Convert raw verifier output into the comparable benchmark final reward.
126
127    ``verifier/reward.json`` may keep continuous diagnostic scores. The
128    externally comparable final reward is binary: every required validation
129    check must pass, otherwise reward/score are 0.
130    """

132    raw_score = _score_or_none(reward.get("raw_score"))
133    raw_reward = _score_or_none(reward.get("raw_reward"))
134    if raw_score is None:
135        raw_score = _score_or_none(reward.get("score", reward.get("reward")))
136    if raw_reward is None:
137        raw_reward = _score_or_none(reward.get("reward", reward.get("score")))
138    raw_passed = bool(reward.get("passed"))
139
140    verifier_completed = verifier_exit in (0, 6)
141    verifier_passed = raw_passed and verifier_completed
142    agent_execution_passed = (
143        agent_exec_returncode in (0, None)
144        or agent_early_terminated
145        or verifier_passed
146    )
147    if agent_exec_returncode in (0, None) or agent_early_terminated:
148        agent_execution_reason = "agent process completed successfully"
149    elif verifier_passed:
150        agent_execution_reason = (
151            f"agent_exec_returncode={agent_exec_returncode}; verifier passed, "
152            "treating agent process status as diagnostic"
153        )
154    else:
155        agent_execution_reason = f"agent_exec_returncode={agent_exec_returncode}"
156
157    if _is_v2_reward_schema(reward):
158        # v2: the raw test.sh check list IS the source of truth. We do NOT
159        # add framework meta-checks (agent_execution / verifier_completed /
160        # script_verifier_or_llm) on top — those gated v1 scoring and made
161        # every report show "x/3" regardless of how many real checks the
162        # task had. Binary passed simply requires every raw check to pass
163        # AND the verifier to have completed (so the raw data is trustable).
164        # An agent crash typically leaves outputs missing and trips raw
165        # checks naturally, so we don't gate on agent_execution explicitly.
166        breakdown = reward.get("checks_breakdown")
167        checks = [
168            {
169                "id": c.get("id"),
170                "passed": bool(c.get("passed")),
171                "reason": c.get("reason"),
172                "check_type": c.get("check_type"),
173            }
174            for c in (breakdown if isinstance(breakdown, list) else [])
175            if isinstance(c, dict)
176        ]
177        passed_checks = sum(1 for c in checks if c.get("passed"))
178        total_checks = len(checks)
179        all_passed = total_checks > 0 and passed_checks == total_checks and verifier_completed
180        check_ratio = round(passed_checks / total_checks, 4) if total_checks else 0.0
181        raw_score = check_ratio
182        raw_reward = check_ratio
183        raw_passed = total_checks > 0 and passed_checks == total_checks
184        llm_check = None
185    else:
186        checks = [
187            {
188                "id": "agent_execution",
189                "passed": agent_execution_passed,
190                "reason": agent_execution_reason,
191            },
192            {
193                "id": "verifier_completed",
194                "passed": verifier_completed,
195                "reason": f"verifier_exit={verifier_exit}",
196            },
197        ]
198        llm_check = _llm_rubric_check(reward, raw_score)
199        if llm_check is not None:
200            checks.append(llm_check)
201        else:
202            checks.append(_script_verifier_check(reward, raw_score, raw_passed))
203        checks.extend(_explicit_validation_checks(reward))
204        hard_checks = reward.get("hard_checks")
205        if isinstance(hard_checks, dict) and "passable" in hard_checks:
206            checks.append(
207                {
208                    "id": "hard_checks_passable",
209                    "passed": bool(hard_checks.get("passable")),
210                    "score": hard_checks.get("score"),
211                    "reason": "; ".join(str(item) for item in hard_checks.get("cap_reasons", []) or []),
212                }
213            )
214        passed_checks = sum(1 for check in checks if check.get("passed"))
215        total_checks = len(checks)
216        all_passed = total_checks > 0 and passed_checks == total_checks
217
218    final_reward["raw_reward"] = raw_reward
219    final_reward["raw_score"] = raw_score
220    final_reward["raw_passed"] = raw_passed
221    if llm_check is not None:
222        final_reward["llm_rubric_score"] = llm_check.get("score")
223        final_reward["llm_rubric_threshold"] = llm_check.get("threshold")
224        final_reward["llm_rubric_passed"] = llm_check.get("passed")
225    final_reward["reward_mode"] = "binary_all_checks"
226    final_reward["validation_checks"] = checks
227    final_reward["check_summary"] = {
228        "total": total_checks,
229        "passed": passed_checks,
230        "failed": total_checks - passed_checks,
231        "all_passed": all_passed,
232    }
233    final_reward["framework_signals"] = {
234        "agent_execution_passed": agent_execution_passed,
235        "agent_execution_reason": agent_execution_reason,
236        "verifier_completed": verifier_completed,
237        "verifier_exit": verifier_exit,
238        "agent_exec_returncode": agent_exec_returncode,
239        "agent_early_terminated": agent_early_terminated,
240    }
241    final_reward["passed"] = all_passed
242    final_reward["reward"] = 1.0 if all_passed else 0.0
243    final_reward["score"] = final_reward["reward"]
244    final_reward.setdefault("verifier_exit", verifier_exit)
245    return final_reward

162    )

164    check.require(

§ 02 · KEY FLOW

关键流程

task config → contract object → runtime materialization → agent execution → verifier/report

从这些源码行可以看到，RealReplicaBench 的流程不是把 Agent 输出丢给一个脚本评分，而是先构造可复现任务世界，再让 harness 运行，最后把 verifier、trajectory 和 report 合并成证据链。

Verifier 页要重点看发布前的双层检查：src/real_replica_bench/reward.py:129 把最终状态交给二值奖励，src/real_replica_bench/reward.py:171 标准化 judge 响应，scripts/validate_release.py:140 遍历公开任务，scripts/validate_release.py:172 检查必须存在的 fixture，tests/test_public_api.py:145 锁住公开 API，tests/test_public_api.py:224 验证评分输出可复现。

§ 03 · EDGE CASES

边界 / 失败模式

▸失败在哪里会被放大

  • 如果任务声明不完整，src/real_replica_bench/reward.py:157 附近的解析会提前中断，而不是让 Agent 在缺环境时跑出伪结果。
  • 如果 mock 或 provider 配置漂移，src/real_replica_bench/reward.py:179 附近的检查会让问题在发布前暴露。
  • 如果 Agent 只完成部分业务动作，最终分数会被 verifier 端硬门禁压成失败，需要回看 trajectory。
