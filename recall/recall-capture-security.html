  26def output_dir(cwd, cfg):
  27    """Resolve the output dir, confined to within cwd, or None if none is safe.
  28
  29    A project-local config is itself untrusted (it ships in the repo), so it must
  30    not be able to redirect writes outside the project via an absolute path or
  31    `..`. A pre-planted symlink at the default `.recall` is just as untrusted: an
  32    escaping config value falls back to `.recall`, but if even that resolves
  33    outside cwd (because `.recall` is a symlink), we return None and refuse to
  34    write rather than land outside the project. Callers treat None as "skip".
  35    """
  36    base = os.path.realpath(cwd)
  37    requested = cfg.get("output_dir", ".recall") or ".recall"
  38    for candidate in (requested, ".recall"):
  39        target = os.path.realpath(os.path.join(cwd, candidate))
  40        if target == base or target.startswith(base + os.sep):
  41            return target
  42    return None

  84def _open_nofollow(path, append):
  85    """Open for writing without following a symlink at the final path component.
  86
  87    Prevents a pre-planted symlink (e.g. history.md -> /etc/something) from
  88    redirecting our writes. Raises OSError (ELOOP) if the target is a symlink.
  89    """
  90    flags = os.O_WRONLY | os.O_CREAT | (os.O_APPEND if append else os.O_TRUNC)
  91    if hasattr(os, "O_NOFOLLOW"):
  92        flags |= os.O_NOFOLLOW
  93    return os.fdopen(os.open(path, flags, 0o644), "w", encoding="utf-8")

 182def git_uncommitted(cwd):
 183    """List of paths with uncommitted changes (porcelain), or []."""
 184    if not _is_git_repo(cwd):
 185        return []
 186    out, seen = [], set()
 187    for line in _git(cwd, ["status", "--porcelain"]).splitlines():
 188        path = line[3:].strip()  # "XY <path>" (rename/copy shows "old -> new")
 189        if " -> " in path:
 190            path = path.split(" -> ", 1)[1].strip()
 191        if path and path not in seen:
 192            seen.add(path)
 193            out.append(path)
 194    return out
