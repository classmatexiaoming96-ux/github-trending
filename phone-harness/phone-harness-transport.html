24def find_window():
25    """{x, y, w, h, id} of the mirroring window in screen points, or None."""
26    wins = Quartz.CGWindowListCopyWindowInfo(
27        Quartz.kCGWindowListOptionOnScreenOnly, Quartz.kCGNullWindowID) or []
28    for w in wins:
29        if w.get("kCGWindowOwnerName") == APP_NAME and w.get("kCGWindowLayer", 1) == 0:
30            b = w["kCGWindowBounds"]
31            if b["Width"] < 100:  # ignore panels/toolbars
32                continue
33            return {"x": b["X"], "y": b["Y"], "w": b["Width"], "h": b["Height"],
34                    "id": int(w["kCGWindowNumber"])}
35    return None

▸截图回退

capture() 先用 window id 截图，失败后激活窗口并按区域截取。src/phone_harness/mirror.py:91、src/phone_harness/mirror.py:105、src/phone_harness/mirror.py:112、src/phone_harness/mirror.py:118 这组行号解释了为什么视频窗口被遮挡或未 composited 时仍有第二条路径。

src/phone_harness/mirror.py · L91-118Python
91def capture(path=None, retries=2):
92    """Capture the mirroring window as a PNG. Returns (path, window_bounds).
93
94    `screencapture -l <id>` grabs only the window (no shadow, even if covered)
95    but fails with "could not create image from window" when the window is not
96    frontmost/composited. So we activate first, and fall back to a region
97    capture of the window rect, which succeeds regardless of window backing
98    (the app is frontmost by then, so nothing occludes the rect).
99    """
100    path = str(path or TMP / "window.png")
101    last = None
102    for attempt in range(retries + 1):
103        win = find_window() or ensure_window()
104        # -l: window-only, ideal when it works
105        ok, last = _run_capture(
106            ["screencapture", "-x", "-o", "-l", str(win["id"]), path], path)
107        if ok:
108            return path, win
109        # not backed — bring it frontmost and retry the region instead
110        activate()
111        win = find_window() or win
112        region = f"{int(win['x'])},{int(win['y'])},{int(win['w'])},{int(win['h'])}"
113        ok, last = _run_capture(
114            ["screencapture", "-x", "-R", region, path], path)
115        if ok:
116            return path, win
117        time.sleep(0.3)
118    raise RuntimeError(f"window capture failed after {retries + 1} tries: {last}")

151def drag(x1, y1, x2, y2, duration=0.35, steps=14):
152    """Touch-drag (what iOS sees as a swipe)."""
153    _focus()
154    _post_mouse(Quartz.kCGEventMouseMoved, x1, y1)
155    time.sleep(0.1)
156    _post_mouse(Quartz.kCGEventLeftMouseDown, x1, y1)
157    for i in range(1, steps + 1):
158        t = i / steps
159        _post_mouse(Quartz.kCGEventLeftMouseDragged,
160                    x1 + (x2 - x1) * t, y1 + (y2 - y1) * t)
161        time.sleep(duration / steps)
162    _post_mouse(Quartz.kCGEventLeftMouseUp, x2, y2)

165def scroll_wheel(dy, x, y, steps=6):
166    """Scroll-gesture at (x, y). Positive dy scrolls content up (finger down)."""
167    _focus()
168    _post_mouse(Quartz.kCGEventMouseMoved, x, y)
169    time.sleep(0.1)

171        ev = Quartz.CGEventCreateScrollWheelEvent(
172            None, Quartz.kCGScrollEventUnitPixel, 1, int(dy / steps))
173        Quartz.CGEventSetLocation(ev, Quartz.CGPointMake(x, y))
174        Quartz.CGEventPost(Quartz.kCGHIDEventTap, ev)
175        time.sleep(0.03)
§ 02 · KEY FLOW

关键流程

capture / OCR / helper → compute screen-point evidence → post Quartz event or raise user-action error → capture again and compare visible state

这个流程的关键是每一步都有可观察证据：src/phone_harness/mirror.py:84、src/phone_harness/ocr.py:48、src/phone_harness/helpers.py:73、src/phone_harness/helpers.py:316、src/phone_harness/admin.py:65 分别覆盖截图、OCR 坐标、screen_info、稳定等待和 doctor 结果。它不像浏览器自动化那样有 DOM，所以截图就是最终断言。
