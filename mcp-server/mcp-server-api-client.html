43export async function call(action: string, params: Record<string, string | number> = {}): Promise<string> {
44  const key = process.env.SVN_API_KEY;
45  if (!key) throw new Error("SVN_API_KEY is not set in the environment.");
46
47  const query = new URLSearchParams({ api_key: key, action, lang: LANG });
48  for (const [name, value] of Object.entries(params)) query.set(name, String(value));
49
50  const response = await fetch(`${BASE}?${query}`, { headers: { Accept: "*/*" } });
51  if (!response.ok) throw new Error(`HTTP ${response.status} from the API.`);
52  return (await response.text()).trim();
53}
54
55/** Calls an action that must answer with `marker`, and returns the reply. */
56export async function expect(marker: string, action: string,
57                             params: Record<string, string | number> = {}): Promise<string> {
58  const reply = await call(action, params);
59  if (!reply.startsWith(marker)) throw new ApiError(reply.split(":")[0], explain(reply));
60  return reply;
61}
62
63export async function json<T>(action: string, params: Record<string, string | number> = {}): Promise<T> {
64  const reply = await call(action, params);
65  try {
66    return JSON.parse(reply) as T;
67  } catch {
68    throw new ApiError(reply.split(":")[0], explain(reply));
69  }
70}

§ 03 · EDGE CASES
