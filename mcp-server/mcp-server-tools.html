71// The catalogue is written in site names, an agent usually holds a URL: "discord.com" has to
72// find "Discord". Strip what the URL adds around the name and keep the raw query as well,
73// because a few entries spell the domain out ("X.com (Twitter)").
74const bareName = (query: string) =>
75  query.toLowerCase().trim()
76    .replace(/^[a-z]+:\/\//, "").replace(/^www\./, "")
77    .replace(/\/.*$/, "").replace(/\.[a-z]{2,}$/, "");
78
79// A short query lands inside unrelated names: "x" sits in the middle of "Maxim". Matching the
80// whole word first keeps those below the site the agent actually asked for.
81const escapeRe = (value: string) => value.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
82const wholeWord = (name: string, needle: string) =>
83  needle.length > 0 && new RegExp(`(^|[^a-z0-9])${escapeRe(needle)}([^a-z0-9]|$)`).test(name);
84
85// Lower is better. Ranking matters because a short query matches dozens of names and the
86// agent picks the first thing it reads.
87const rank = (service: Service, query: string, bare: string) => {
88  const id = service.id.toLowerCase();
89  const name = service.name.toLowerCase();
90  if (id === query || id === bare) return 0;
91  if (name === query || name === bare) return 1;
92  if (wholeWord(name, query)) return 2;
93  if (wholeWord(name, bare)) return 3;
94  if (name.includes(query)) return 4;
95  if (name.startsWith(bare)) return 5;
96  if (name.includes(bare)) return 6;
97  return -1;
98};

103  return { service: service.id, name: service.name, price: service.price,
104           online: service.quantity, deliveredPercent: measured > 0 ? measured : null };
