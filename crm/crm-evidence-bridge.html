93const CEILING = 0.99;

95const CONTRADICTED = 0.45;

97export const BAND_FLOOR = { VERIFIED: 0.85, PROBABLE: 0.55, POSSIBLE: 0.3 };
98
99export function scoreEvidence(evidence: Evidence[]): Scored {
100	if (evidence.length === 0) {
101		return {
102			score: 0,
103			band: null,
104			hasPrimary: false,
105			rationale: "No evidence.",
106		};
107	}
108
109	const contradicted = evidence.some((item) => item.kind === "contradiction");
110	const hasPrimary = evidence.some((item) => WEIGHTS[item.kind].primary);
111
112	const combined = evidence.reduce(
113		(remaining, item) => remaining * (1 - WEIGHTS[item.kind].weight),
114		1,
115	);
116
117	let score = Math.min(CEILING, 1 - combined);
118	if (contradicted) score = Math.min(score, CONTRADICTED);
119
120	return {
121		score,
122		band: bandFor(score, hasPrimary),
123		hasPrimary,
124		rationale: rationaleFor(evidence, contradicted, hasPrimary),
125	};
126}
127
128export function bandFor(score: number, hasPrimary: boolean): FactBand | null {
129	if (score >= BAND_FLOOR.VERIFIED && hasPrimary) return FactBand.VERIFIED;
130	if (score >= BAND_FLOOR.PROBABLE) return FactBand.PROBABLE;
131	if (score >= BAND_FLOOR.POSSIBLE) return FactBand.POSSIBLE;
132	return null;

§ 02 · KEY FLOW

关键流程

agent observes claim → record_fact tool → scoreEvidence produces band → fact transaction stores APPLIED or PROPOSED → UI transcript shows source markers → conversation save preserves continuation

这个流程把可信度变成产品状态：packages/db/prisma/schema.prisma:265 的 ContactFact 存 evidence 与 status，apps/app/components/crm/agent-panel.tsx:400 的 Sources 组件把来源展示给代表。

§ 03 · EDGE

边界 / 失败模式

  • apps/agent/agent/lib/evidence.ts:117 设置 CEILING，测试 apps/agent/test/evidence.spec.ts:75 确认永不声称绝对确定。
  • apps/app/lib/agent-bridge.ts:20 没有 secret 就拒绝 mint，测试 apps/app/test/agent-bridge.spec.ts:74 覆盖。
  • apps/app/components/crm/agent-panel.tsx:253 在旧问题仍 working 时锁住下一问，避免前端把同一 session 搞乱。
