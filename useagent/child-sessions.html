241  const eventsByExecution = new Map<string, Map<string, CanonicalThreadEvent>>();
242
243  return {
244    ingest(event) {
245      if (event.kind === "child.started") {
246        childIds.add(event.childId);
247        const parentId = event.parentChildId
248          ?? event.identity.nativeParentSessionId
249          ?? (event.identity.nativeSessionId !== undefined
250            && event.identity.nativeSessionId !== event.childId
251            ? event.identity.nativeSessionId
252            : event.runId);
253        delegationEdges.set(`${parentId}:${event.childId}`, { parentId, childId: event.childId });
254      } else if (event.kind === "child.updated" || event.kind === "child.completed") {
255        childIds.add(event.childId);
256      }
257      const executionId = event.identity.nativeSessionId ?? event.runId;
258      const existing = eventsByExecution.get(executionId) ?? new Map<string, CanonicalThreadEvent>();

260      if (
261        previous === undefined ||
262        event.revision > previous.revision ||
263        (event.revision === previous.revision && event.deliverySeq > previous.deliverySeq)
264      ) {
265        existing.set(event.eventId, event);
266      }
267      eventsByExecution.set(executionId, existing);
268    },
269    snapshot() {
270      return {
271        executionIds: [...eventsByExecution.keys()],
272        childIds: [...childIds],
273        delegationEdges: [...delegationEdges.values()],
274        eventsByExecution: new Map(
