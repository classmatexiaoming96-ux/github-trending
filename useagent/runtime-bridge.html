OpenCode server 是可复用 resident process

先探活再启动

`ensureServer` 先准备 workdir、拿 sandbox preview link，再探测 `/global/health`。如果已有 healthy resident process，它直接复用；只有不健康时才删除 session、创建 `skynet-opencode-serve` 并执行 `opencode serve`。 backend/src/engines/opencode-server.ts:252 backend/src/engines/opencode-server.ts:267 backend/src/engines/opencode-server.ts:279 backend/src/engines/opencode-server.ts:283

252async function ensureServer(
253  sandbox: SandboxHandle,
254  npx: boolean,
255  signal: AbortSignal,
256  secretSourceCommand = sandboxSecretSourceCommand(),
257): Promise<OpenCodeRuntimeServer> {
258  const bin = npx ? `npx -y opencode-ai@${OPENCODE_VERSION}` : "opencode";
259  const homeResult = await sandbox.process.executeCommand(
260    'mkdir -p ~/work && printf %s "$HOME"',
261    undefined,
262    undefined,
263    15,
264  );
265  if ((homeResult.exitCode ?? 1) !== 0) throw new Error("opencode workspace preparation failed");
266  const home = homeResult.result?.trim() || "/home/daytona";
267  const link = await sandbox.getPreviewLink(SERVE_PORT);
268  const baseUrl = link.url.replace(/\/+$/, "");
269  const token = link.token ?? "";
270  const server = { baseUrl, token, workdir: `${home}/work` };
271
272  // A healthy resident process survives turns and is always reused: the liveness
273  // probe here already proves it is serving, so return immediately and skip the
274  // readiness poll's redundant first probe (perf plan Phase 1 item 4 - skip probes
275  // once a prior step proves liveness). Only when the sandbox was stopped/restarted
276  // (probe not 2xx) do we recreate Daytona's dedicated background session:
277  // executeCommand is synchronous even with shell `&`, whereas an async session
278  // command is Daytona's supported long-lived-process primitive.
279  const initialStatus = await opencodeHealthStatus(server, signal);
280  if (initialStatus !== null && initialStatus >= 200 && initialStatus < 300) {
281    return server;
282  }
283  await sandbox.process.deleteSession(SERVER_PROCESS_SESSION).catch(() => {});
284  await sandbox.process.createSession(SERVER_PROCESS_SESSION);
285  await sandbox.process.executeSessionCommand(
286    SERVER_PROCESS_SESSION,
287    {
288      command: `${secretSourceCommand} && cd ${shq(`${home}/work`)} && exec ${bin} serve --hostname 0.0.0.0 --port ${SERVE_PORT}`,
289      runAsync: true,
290      suppressInputEcho: true,
291    },
292    30,
293  );
294
295  const deadline = Date.now() + 120_000;
296  let lastStatus: number | null = null;
297  // Bounded exponential polling (perf plan Phase 1): a server that is ready in
298  // 200ms is seen in ~200ms instead of at the next full-second tick; the overall
299  // deadline is unchanged.
300  let pollDelay: number | null = null;
301  while (Date.now() < deadline && !signal.aborted) {
302    lastStatus = await opencodeHealthStatus(server, signal);
303    if (lastStatus !== null && lastStatus >= 200 && lastStatus < 300) {
304      return server;
305    }
306    const delay = nextPollDelayMs(pollDelay);
307    pollDelay = delay;
308    await new Promise((resolve) => setTimeout(resolve, delay));
309  }
310  if (signal.aborted) throw new Error("opencode run aborted (timeout)");
311  let logs: { output?: string; stderr?: string; stdout?: string } | null = null;
312  try {
313    const session = await sandbox.process.getSession(SERVER_PROCESS_SESSION);
314    const command = session.commands.at(-1);
315    if (command?.id) {
316      logs = await sandbox.process.getSessionCommandLogs(SERVER_PROCESS_SESSION, command.id);
317    }
318  } catch {
319    // Readiness already failed. Logs are diagnostic only and must not mask the
320    // stable, redacted error below.
321  }
322  const safeTail = (logs?.output ?? logs?.stderr ?? logs?.stdout ?? "")
323    .replace(/v1\.[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+/g, "<capability>")
324    .trim();
325  throw new Error(
326    `opencode serve failed readiness${lastStatus ? ` (HTTP ${lastStatus})` : ""}: ${truncate(safeTail, 200)}`,
327  );
328}

363async function opencodeHealthStatus(
364  server: OpenCodeRuntimeServer,
365  signal: AbortSignal,
366): Promise<number | null> {
367  try {
368    const response = await fetch(`${server.baseUrl}/global/health`, {
369      headers: authHeaders(server.token),
370      signal: AbortSignal.any([signal, AbortSignal.timeout(5_000)]),
371    });
372    const status = response.status;
373    await response.body?.cancel().catch(() => {});
374    return status;
375  } catch {
376    return null;
377  }

387  return status !== null && status >= 200 && status < 300 ? cached : null;
388}

Turn 调度把 runtime snapshot 当权威事实

调度前先确认 provider ready

runtime adapter 先执行 provider readiness barrier，失败就重启 runtime 并 invalidate access，再次验证不通过才抛错。这个策略让 provider 配置异步落盘带来的 race 能被显式处理。 backend/src/engines/runtime-adapter.ts:90 backend/src/engines/runtime-adapter.ts:101 backend/src/engines/runtime-adapter.ts:111 backend/src/engines/runtime-adapter.ts:123

通用 provider readiness barrierbackend/src/engines/runtime-adapter.ts:90
90export async function ensureRuntimeProviderReadyForTurn(input: {
91  readonly sandbox: SandboxHandle;
92  readonly signal: AbortSignal;
93  readonly readiness: RuntimeProviderReadiness;
94  readonly barrierDeadlineMs: number;
95  readonly verifyDeadlineMs: number;
96  readonly providerLabel: string;
97  readonly dependencies?: RuntimeProviderBarrierDependencies;
98}): Promise<void> {
99  const dependencies = input.dependencies ?? runtimeProviderBarrierDependencies;
100  if (
101    await dependencies.awaitReady(
102      input.sandbox,
103      input.signal,
104      input.barrierDeadlineMs,
105      input.readiness,
106    )
107  ) {
108    return;
109  }
110  input.signal.throwIfAborted();
111  await dependencies.restart(input.sandbox, input.signal);
112  dependencies.invalidateAccess(input.sandbox);
113  input.signal.throwIfAborted();
114  if (
115    !(await dependencies.awaitReady(
116      input.sandbox,
117      input.signal,
118      input.verifyDeadlineMs,
119      input.readiness,
120    ))
121  ) {
122    input.signal.throwIfAborted();
123    throw new Error(`${input.providerLabel} runtime did not become ready after restart`);
124  }
125}

403  state: { readonly publishedText: string; readonly finalText: string },
