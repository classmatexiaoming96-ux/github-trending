dispatchMany 不让一个任务拖垮批次

`dispatchMany` 先校验任务数不超过 100，再用 `mapWithConcurrency` 控并发；每个任务返回 ok/error outcome，而不是整批 reject。这样 CLI fan 可以把失败任务写进 JSONL 结果而不是丢失所有成功 run。 packages/agent-client/src/fleet.ts:161 packages/agent-client/src/fleet.ts:199 packages/agent-client/src/fleet.ts:319 packages/agent-client/src/fleet.ts:333

mapWithConcurrency 保序控并发packages/agent-client/src/fleet.ts:199
199/** Run `fn` over `items` with at most `concurrency` in flight, preserving input order. */
200async function mapWithConcurrency<T, R>(
201  items: readonly T[],
202  concurrency: number,
203  fn: (item: T, index: number) => Promise<R>,
204): Promise<R[]> {
205  const results = new Array<R>(items.length);
206  let cursor = 0;
207  const worker = async (): Promise<void> => {
208    for (;;) {
209      const index = cursor++;
210      if (index >= items.length) return;
211      results[index] = await fn(items[index]!, index);
212    }
213  };
214  const size = Math.min(Math.max(1, concurrency), items.length);
215  await Promise.all(Array.from({ length: size }, () => worker()));
216  return results;
217}

61export async function fanCommand(client: FleetClient, args: FanArgs, io: CommandIO): Promise<number> {
62  const tasks = parseTasksJsonl(await io.readFile(args.file));
63  io.err(`dispatching ${tasks.length} task(s), parallelism ${args.parallel}${args.qc ? ", QC on" : ""}...`);
64
65  const outcomes = await client.dispatchMany(tasks, { concurrency: args.parallel });
66  const results: FanResultLine[] = await Promise.all(
67    outcomes.map(async (outcome): Promise<FanResultLine> => {
68      if (!outcome.ok) {
69        return { prompt: outcome.task.prompt, runId: null, status: "dispatch_error", answer: "", url: null, error: outcome.error };
70      }
71      let settled: Awaited<ReturnType<FleetClient["awaitSettled"]>>;
72      try {
73        settled = await client.awaitSettled(outcome.run.runId);
74      } catch (error) {
75        return {
76          prompt: outcome.task.prompt,
77          runId: outcome.run.runId,
78          status: "settle_error",
79          answer: "",
80          url: outcome.run.url,
81          error: errorMessage(error),
82        };
83      }
84      const base: FanResultLine = {
85        prompt: outcome.task.prompt,
86        runId: outcome.run.runId,
87        status: settled.status,
88        answer: settled.answer,
89        url: settled.url,
90      };
91      if (!args.qc) return base;
92      // QC is a real reply run recorded in the SAME thread as the work it judges.
93      try {
94        const verified = await client.verify(outcome.run.runId, args.qc);
95        return { ...base, verdict: verified.verdict };
96      } catch (error) {
97        return {
98          ...base,
99          status: "verification_error",
100          verdict: "unknown",
101          error: errorMessage(error),
102        };
103      }
104    }),
105  );
106
107  const jsonl = serializeResults(results);
108  if (args.out) {
109    await io.writeFile(args.out, jsonl);
110    io.err(`wrote ${results.length} result(s) to ${args.out}`);
111  } else {
112    io.out(jsonl.trimEnd());
113  }
114  io.err("");
115  io.err(formatFanSummary(results));
116
117  const allGood = results.every(
118    (r) => r.status === "completed" && r.error === undefined && r.verdict !== "fail",
119  );
120  return allGood ? 0 : 1;
