  60 def build_request_messages(
  61     messages: list[dict[str, Any]],
  62     *,
  63     rolling_breakpoint: bool = True,
  64 ) -> list[dict[str, Any]]:
  65     """The outgoing request's messages: a request-shaped copy of ``messages`` with the
  66     rolling cache breakpoint on the newest persisted content block.
  67 
  68     Within a turn the system blocks are constant, so the marker makes each round's prior
  69     rounds — the long tool results especially — a cache read instead of a reprocess. Two
  70     cases skip it. A bare first call (one message) would write an entry a one-shot
  71     session never reads, and the next call marks a later block whose span carries the
  72     first message anyway. And the caller passes ``rolling_breakpoint=False`` on rounds
  73     whose ``tool_choice`` is not ``auto``, because ``tool_choice`` keys the messages
  74     span: an entry written under a forced round is unreadable by the auto rounds that
  75     follow.
  76 
  77     A turn that closed on a presentation round leaves a tool_result message followed by
  78     the next user message; the two go out as one user message, tool_result blocks first.
  79 
  80     Applied per call, to the outgoing request only: the returned list shallow-copies the
  81     touched message and blocks, strips the marker any earlier call placed, and never
  82     mutates the host's persisted history. String content is lifted into a one-block list
  83     because ``cache_control`` lives on content blocks.
  84     """
  85     if not messages:
  86         return []
  87 
  88     def without_marker(message: dict[str, Any]) -> dict[str, Any]:
  89         content = message.get("content")
  90         if not isinstance(content, list) or not any(
  91             isinstance(b, dict) and "cache_control" in b for b in content
  92         ):
  93             return message
  94         return message | {
  95             "content": [
  96                 {k: v for k, v in b.items() if k != "cache_control"} if isinstance(b, dict) else b
  97                 for b in content
  98             ]
  99         }
 100 
 101     def blocks(raw: Any) -> list[Any]:
 102         return [{"type": "text", "text": raw}] if isinstance(raw, str) else list(raw or [])
 103 
 104     request: list[dict[str, Any]] = []
 105     for message in messages:
 106         message = without_marker(message)
 107         if request and message.get("role") == "user" and request[-1].get("role") == "user":
 108             request[-1] = request[-1] | {
 109                 "content": blocks(request[-1].get("content")) + blocks(message.get("content"))
 110             }
 111         else:
 112             request.append(message)
 113     if not rolling_breakpoint or len(request) < 2:
 114         return request
 115     content = blocks(request[-1].get("content"))
 116     if content and isinstance(content[-1], dict):
 117         content[-1] = {**content[-1], "cache_control": {"type": "ephemeral"}}
 118         request[-1] = request[-1] | {"content": content}
 119     return request
