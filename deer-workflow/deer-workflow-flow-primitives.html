78export async function pipeline<TOriginal>(
79  items: readonly TOriginal[],
80  ...stages: RuntimePipelineStage<TOriginal>[]
81): Promise<Array<unknown | null>> {
82  return Promise.all(
83    Array.from(items, async (original, index) => {
84      let value: unknown = original;
85
86      try {
87        for (const stage of stages) {
88          value = await stage(value as never, original, index);
89        }
90        return value;
91      } catch {
92        return null;
93      }
94    }),
95  );
96}

§ 03 · EDGE CASES

边界 / 失败模式

  • 源码把错误路径显式化，但调用方仍要决定 null、throw 和 retry 的业务语义。
  • CLI adapter 依赖宿主机安装对应命令，测试中通过 stub 覆盖，生产中还需要版本治理。
  • 事件流是同步 emitter，监听器自身不能做阻塞重活。
