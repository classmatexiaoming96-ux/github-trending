132def find_manifest(source: Path) -> Path:
133    source = source.expanduser().resolve()

162def safe_project_path(root: Path, relative: str) -> Path:

173def build_image_map(root: Path) -> Dict[str, str]:
174    image_map: Dict[str, str] = {}
175    total = 0
176    for path in sorted(root.rglob("*")):
177        if not path.is_file() or path.suffix.lower() not in IMAGE_MIME:
178            continue
179        size = path.stat().st_size
180        if size > MAX_IMAGE_BYTES:
181            log(f"skip local image over 20 MiB: {path.relative_to(root)}")
182            continue
183        if total + size > MAX_EMBEDDED_MEDIA_BYTES:
184            raise ExportError(
185                "local image payload exceeds 200 MiB; reduce media size or use remote URLs"
186            )
187        data = base64.b64encode(path.read_bytes()).decode("ascii")
188        rel = path.relative_to(root).as_posix()
189        image_map[rel] = f"data:{IMAGE_MIME[path.suffix.lower()]};base64,{data}"
190        total += size
191    if image_map:
192        log(f"prepared {len(image_map)} local image resource(s), {total} bytes")
193    return image_map

196def build_payload(manifest: Path) -> Dict[str, Any]:
197    manifest_text, manifest_data = read_yaml_mapping(manifest)
198    if manifest_data.get("version") != "v2":
199        raise ExportError("local PPTX export currently requires PPTD version: v2")
200    page_paths = manifest_data.get("pages")
201    if not isinstance(page_paths, list) or not page_paths:
202        raise ExportError("PPTD manifest must contain a non-empty pages list")
203
204    root = manifest.parent.resolve()
205    pages: List[Dict[str, str]] = []
206    for entry in page_paths:
207        page_path = safe_project_path(root, entry)
208        if not page_path.is_file():
209            raise ExportError(f"missing page file: {entry}")
210        page_text, page_data = read_yaml_mapping(page_path)
211        if not isinstance(page_data.get("elements"), list):
212            raise ExportError(f"page elements must be an array: {entry}")
213        pages.append({"path": str(entry), "content": page_text})
214
215    title = str(manifest_data.get("title") or manifest.stem)
216    return {
217        "id": f"local-export-{uuid.uuid4().hex}",
218        "title": title,
219        "manifestPath": manifest.name,
220        "manifestContent": manifest_text,
221        "pages": pages,
222        "imageMap": build_image_map(root),
223    }

376def replace_transition(slide_xml: bytes, transition: str) -> bytes:
377    text = slide_xml.decode("utf-8")
378    pattern = re.compile(
379        r"<p:transition\b[^>]*(?:/>|>.*?</p:transition>)", re.DOTALL
380    )
381    text = pattern.sub("", text)
382    if transition == "none":
383        return text.encode("utf-8")
384
385    # CT_Slide requires transition as a direct child after cSld/clrMapOvr and
386    # before timing/extLst. Searching for the first p:extLst is incorrect:
387    # shapes may contain their own nested extLst inside cSld, causing Office to
388    # ignore a transition inserted there.
389    color_map = re.search(
390        r"<p:clrMapOvr\b[^>]*(?:/>|>.*?</p:clrMapOvr>)", text, re.DOTALL
391    )
392    common_slide = re.search(
393        r"<p:cSld\b[^>]*(?:/>|>.*?</p:cSld>)", text, re.DOTALL
394    )
395    anchor = color_map or common_slide
396    if anchor is None:
397        raise ExportError("slide XML has no cSld/clrMapOvr insertion anchor")
398    position = anchor.end()
399    return (text[:position] + FADE_TRANSITION_XML + text[position:]).encode("utf-8")
