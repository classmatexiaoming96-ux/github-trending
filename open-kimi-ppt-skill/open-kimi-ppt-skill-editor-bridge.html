编辑器桥接与保存事务 深度解析

这一页专门拆 File System Access、PPTD manifest 载入、图片解析、远端 iframe RPC 和保存事务。核心问题是：怎样让远端编辑器能改本地 PPTD，同时不越权、不乱序、不吞错。

▸图片解析只返回 data URL 或空字符串

图片资源解析在 lib/editor-app.js:442 先允许 data/http/blob，其他路径走 indexed file。lib/editor-app.js:486 到 lib/editor-app.js:495 读取文件并按 lastModified/size 缓存 data URL；lib/editor-app.js:498 会记录 miss 数量但返回空字符串而非中断整个 deck。这种降级适合编辑器预览，避免一张丢失图片阻断所有页面。

442function resolveIndexedPath(requestedPath) {
443  if (typeof requestedPath !== "string" || !requestedPath.trim()) return null;
444  if (/^(?:data:image\/|https?:\/\/|blob:)/i.test(requestedPath)) return requestedPath;
445  let candidate = requestedPath.replace(/^file:\/\/+/, "").replaceAll("\\", "/");
446  try {
447    candidate = decodeURIComponent(candidate);
448  } catch {
449    // Preserve literal percent signs.
450  }
451
452  const directCandidates = [];
453  try {
454    directCandidates.push(normalizeRelativePath(candidate));
455  } catch {
456    const parts = candidate.split("/").filter(Boolean);
457    for (let index = 0; index < parts.length; index += 1) {
458      try {
459        directCandidates.push(normalizeRelativePath(parts.slice(index).join("/")));
460      } catch {
461        // Keep looking for a safe suffix.
462      }
463    }
464  }
465  try {
466    directCandidates.push(joinDeckPath(state.manifestDirectory, candidate));
467  } catch {
468    // Absolute paths may still match a safe indexed suffix below.
469  }
470  for (const path of directCandidates) if (state.fileIndex.has(path)) return path;
471  for (const path of state.fileIndex.keys()) {
472    if (directCandidates.some((candidatePath) => path.endsWith(`/${candidatePath}`) || path === candidatePath)) return path;
473  }
474  return null;
475}

486async function resolveImage(requestedPath) {
487  if (/^(?:data:image\/|https?:\/\/|blob:)/i.test(requestedPath)) return requestedPath;
488  const path = resolveIndexedPath(requestedPath);
489  if (!path) return "";
490  const entry = state.fileIndex.get(path);
491  const file = await entry.getFile();
492  if (file.size > MAX_IMAGE_BYTES) return "";
493  const cacheKey = `${path}:${file.lastModified}:${file.size}`;
494  if (!state.imageCache.has(cacheKey)) state.imageCache.set(cacheKey, fileToDataUrl(file));
495  return state.imageCache.get(cacheKey);
496}

507  const parts = normalizeRelativePath(path).split("/");

556async function persistChanges(payload = {}) {
557  const changes = Array.isArray(payload.changes) ? payload.changes : [];
558  if (changes.length === 0) return { fileContent: payload.fileContent, lastModifiedTime: Date.now() };
559  if (changes.length > MAX_CHANGE_COUNT) throw new Error(`一次保存不能超过 ${MAX_CHANGE_COUNT} 个文件`);
560
561  const normalized = changes.map((change) => {
562    const path = assertWritableChangePath(state.manifestDirectory, change.path);
563    const operation = change.operate === "delete" ? "delete" : "put";
564    const content = operation === "put" ? String(change.content ?? "") : null;
565    if (content && new Blob([content]).size > MAX_TEXT_BYTES) throw new Error(`保存内容超过 20 MiB：${path}`);
566    return { path, operation, content };
567  });
568
569  if (state.source === "demo") {
570    for (const change of normalized) {
571      if (change.operation === "delete") state.memoryFiles.delete(change.path);
572      else state.memoryFiles.set(change.path, change.content);
573    }
574    setSaveState("刚刚保存到内存", "memory");
575    addActivity(`已在内存保存 ${normalized.length} 项变更`, "success");
576    return { fileContent: payload.fileContent, lastModifiedTime: Date.now() };
577  }
578
579  if (state.readOnlyFallback || !state.directoryHandle) {
580    throw new Error("当前是只读兼容模式，请用支持文件夹授权的 Chromium 浏览器打开");
581  }
582
583  const permission = await state.directoryHandle.queryPermission({ mode: "readwrite" });
584  if (permission !== "granted") throw new Error("项目文件夹的写入权限已经失效");
585
586  setSaveState("正在保存…", "saving");
587  const backups = new Map();
588  try {
589    for (const change of normalized) backups.set(change.path, await readBackup(change.path));
590    for (const change of normalized) {
591      if (change.operation === "delete") await deleteFile(change.path);
592      else await writeFile(change.path, change.content);
593    }
594  } catch (error) {
595    await rollbackChanges(backups).catch((rollbackError) => {
596      addActivity(`回滚未完全成功：${rollbackError.message}`, "error");
597    });
598    throw error;
599  }
600
601  setSaveState("刚刚已保存", "saved");
602  addActivity(`已保存 ${normalized.length} 个文件变更`, "success");
