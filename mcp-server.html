§ 00 · Source-level Observatory

mcp-server 给 Agent 一支临时手机号

SV Number MCP server 把商业短信验证码 API 包成 9 个 MCP stdio 工具：Agent 可以查国家、查服务、下单号码、等待验证码、完成或取消 activation，还能本地计算 TOTP。这个项目值得拆，因为它把“不稳定、文本 marker 风格的外部 API”翻译成 Agent 可理解、可恢复的工具契约。

§ 00 · OVERVIEW

77    .replace(/\/.*$/, "").replace(/\.[a-z]{2,}$/, "");

80// whole word first keeps those below the site the agent actually asked for.
81const escapeRe = (value: string) => value.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
82const wholeWord = (name: string, needle: string) =>
83  needle.length > 0 && new RegExp(`(^|[^a-z0-9])${escapeRe(needle)}([^a-z0-9]|$)`).test(name);

85// Lower is better. Ranking matters because a short query matches dozens of names and the
86// agent picks the first thing it reads.
87const rank = (service: Service, query: string, bare: string) => {
88  const id = service.id.toLowerCase();
89  const name = service.name.toLowerCase();
90  if (id === query || id === bare) return 0;
91  if (name === query || name === bare) return 1;
92  if (wholeWord(name, query)) return 2;
93  if (wholeWord(name, bare)) return 3;
94  if (name.includes(query)) return 4;
95  if (name.startsWith(bare)) return 5;
96  if (name.includes(bare)) return 6;
97  return -1;
98};

43export async function call(action: string, params: Record<string, string | number> = {}): Promise<string> {
44  const key = process.env.SVN_API_KEY;
45  if (!key) throw new Error("SVN_API_KEY is not set in the environment.");
46
47  const query = new URLSearchParams({ api_key: key, action, lang: LANG });
48  for (const [name, value] of Object.entries(params)) query.set(name, String(value));
49
50  const response = await fetch(`${BASE}?${query}`, { headers: { Accept: "*/*" } });
51  if (!response.ok) throw new Error(`HTTP ${response.status} from the API.`);
52  return (await response.text()).trim();
53}

25export function totp(secret: string, options: { digits?: number; period?: number;
26                                                algorithm?: string; at?: number } = {}) {
27  const digits = options.digits ?? 6;
28  const period = options.period ?? 30;
29  const algorithm = (options.algorithm ?? "sha1").toLowerCase();
30  const seconds = options.at ?? Math.floor(Date.now() / 1000);
31  const counter = Math.floor(seconds / period);
32
33  const message = Buffer.alloc(8);
34  message.writeBigUInt64BE(BigInt(counter));
35  const digest = createHmac(algorithm, decodeBase32(secret)).update(message).digest();
36
37  const offset = digest[digest.length - 1] & 0x0f;
38  const binary = ((digest[offset] & 0x7f) << 24) | (digest[offset + 1] << 16)
39               | (digest[offset + 2] << 8) | digest[offset + 3];
40  const code = (binary % 10 ** digits).toString().padStart(digits, "0");
41
42  return { code, secondsLeft: period - (seconds % period) };
43}
