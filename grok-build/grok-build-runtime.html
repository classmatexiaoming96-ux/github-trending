126/// Resolve the two post-reconnect restore outcomes from the per-agent
127/// `session/load` results.
128///
129/// - `all_restored` (AND across every reloaded tab, plus `init_ok`) drives the
130///   user-facing toast: it reports whether the WHOLE reconnect came back.
131/// - `active_restored` is per-agent: the ACTIVE tab's OWN reload succeeded. It
132///   gates that tab's post-reconnect queue drain. Gating the drain on
133///   `all_restored` would let one failed background tab strand prompts queued
134///   on a healthy active tab — the drain (`dispatch_drain_queue`) only ever
135///   touches the active agent, so a background failure has no bearing on it.
136///
137/// `loads` maps each reloaded agent to `(success, running_prompt_id)`; an agent
138/// in `pending_agent_ids` but absent from `loads` is treated as failed
139/// (mirrors the `unwrap_or((false, _))` at the finalize site).
140fn reconnect_restore_outcome(
141    init_ok: bool,
142    pending_agent_ids: &[super::agent::AgentId],
143    loads: &std::collections::HashMap<super::agent::AgentId, (bool, Option<String>)>,
144    active_agent_id: Option<super::agent::AgentId>,
145) -> (bool, bool) {
146    let load_ok = |id: &super::agent::AgentId| -> bool { loads.get(id).is_some_and(|(ok, _)| *ok) };
147    let all_restored = init_ok && pending_agent_ids.iter().all(load_ok);
148    let active_restored = init_ok
149        && active_agent_id.is_some_and(|aid| pending_agent_ids.contains(&aid) && load_ok(&aid));
150    (all_restored, active_restored)

153/// Compute the folder-trust verdict for the session cwd and seed
154/// [`AppView::trust_state`]. Pager-side mirror of the agent's resolve: read the
155/// local store, scan for repo-local code-exec config, and run the pure
156/// [`decide`](xai_grok_workspace::folder_trust::decide) precedence.
157///
158/// `TrustOutcome::Prompt` (interactive + untrusted + repo configs present)
159/// becomes `TrustState::Pending` (show the question); everything else becomes
160/// `TrustState::Done`. The feature-off fast path (kill-switch / opt-out /
161/// local build) short-circuits before any I/O.
162fn seed_trust_state(
163    app: &mut AppView,
164    remote: Option<&xai_grok_shell::util::config::RemoteSettings>,
165) {
166    use std::io::IsTerminal;
167    use xai_grok_workspace::folder_trust::{
168        TrustOutcome, decide, decide_inputs_with_interactive, feature_enabled,
169    };
170    use xai_grok_workspace::trust::workspace_key;
171
172    let feature = feature_enabled(remote);
173    if !feature {
174        app.trust_state = TrustState::Done;
175        return;
176    }
177
178    // The cwd the user launched in == the process cwd == `app.cwd` (set at
179    // construction), matching the `--trust` grant's `std::env::current_dir()`.
180    let cwd = app.cwd.clone();
181    let key = workspace_key(&cwd);
182    // Reuse the canonical gather (store trust + repo-config scan) but pass the
183    // pager's stdin-only interactivity: the TUI prompts via the rendered
184    // question + crossterm keyboard, NOT stderr (the pager redirects native
185    // stderr at startup, so the engine's `stdin && stderr` would be false here
186    // and the question would never show). TTY stdin => user can answer;
187    // otherwise fail closed (no prompt).
188    let inputs = decide_inputs_with_interactive(&cwd, &key, std::io::stdin().is_terminal());
189    app.trust_state = match decide(feature, &inputs) {
190        TrustOutcome::Prompt => TrustState::Pending { workspace: key },
191        TrustOutcome::Trusted | TrustOutcome::Untrusted => TrustState::Done,
192    };
193}

195/// Suspend the inline TUI, run a blocking child that takes over the tty
196/// (`$EDITOR`, `$PAGER`, …), then restore. Shared by the editor and transcript
197/// suspend paths so the subtle reader-park + raw-mode + alt-screen handoff lives
198/// in one place.
199///
200/// The reader thread is parked first so the child (which inherits this tty)
201/// keeps every keystroke instead of racing the reader; on return, buffered
202/// terminal query replies and any pre-park keystroke are drained before the
203/// reader resumes.
204///
205/// The frame **writer** thread is then drained (bounded) before the child
206/// starts: frames are written to the tty asynchronously, and the frame that
207/// armed this suspend (e.g. minimal's final `/transcript` pump slice) is
208/// typically queued microseconds before we get here. Un-drained, those bytes
209/// race the child's own output — they can land on the child's alternate
210/// screen (so the main screen never receives them; on an inline viewport a
211/// commit's scroll then leaves every following row off by its height) or tear
212/// around the alt-screen switch, printing escape fragments (`[`…) that the
213/// renderer's diff can't see and thus never repairs.
214///
215/// In minimal mode the physical cursor is probed (`ESC[6n`) right before the
216/// child runs and right after it exits — while the reader thread is still
217/// parked, so the replies can't be stolen. The returned "cursor after the
218/// child, iff it moved" tells the caller whether the child *printed to the
219/// main screen* (cat-style pager: cursor left below its output) or *restored
220/// it* (less-style alt-screen pager: `rmcup` puts the cursor back exactly
221/// where it was): the caller re-anchors the live region below the new output
222/// in the first case and repaints in place in the second.
223fn suspend_for_child(
224    screen_mode: crate::app::ScreenMode,
225    writer_sync: &crate::render::draw::WriterSync,
226    input_paused: &std::sync::atomic::AtomicBool,
227    reader_parked: &std::sync::atomic::AtomicBool,
228    input_rx: &mut tokio::sync::mpsc::UnboundedReceiver<crossterm::event::Event>,
229    run_child: impl FnOnce(),
230) -> Option<(u16, u16)> {
231    use std::sync::atomic::Ordering;
232    // Pause the reader thread, then wait for a FRESH park (reader provably out of
233    // crossterm) so the main thread is the sole poll/read caller; bounded so a
234    // dead reader can't hang us.
235    input_paused.store(true, Ordering::Release);
236    reader_parked.store(false, Ordering::Release);
237    let park_deadline = std::time::Instant::now() + Duration::from_millis(500);
238    while !reader_parked.load(Ordering::Acquire) && std::time::Instant::now() < park_deadline {
239        std::thread::sleep(Duration::from_millis(5));
240    }
241    // Every queued frame must be ON the tty before the child takes it (and, in
242    // fullscreen, before LeaveAlternateScreen below — mirroring teardown's
243    // "no late frame after LeaveAlternateScreen" drain). Bounded like the
244    // reader park so a wedged pty can't hang the suspend.
245    if !writer_sync.wait_drained(Duration::from_millis(750)) {
246        tracing::warn!("suspend: frame writer not drained within 750ms; proceeding");
247    }
248    // Pre-child cursor probe (minimal only — minimal's startup already proved
249    // this terminal answers CPR). Reader is parked, so the reply is ours.
250    let pre_cursor = screen_mode
251        .is_minimal()
252        .then(|| crossterm::cursor::position().ok())
253        .flatten();
254    if screen_mode.is_fullscreen() {
255        xai_grok_shell::util::with_locked_stderr(|stderr| {
256            let _ = crossterm::execute!(stderr, crossterm::terminal::LeaveAlternateScreen);
257        });
258    }
259    let _ = crossterm::terminal::disable_raw_mode();
260    run_child();
