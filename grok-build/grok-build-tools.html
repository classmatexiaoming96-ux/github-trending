6/// Capability mode applied to a session's toolset.
7///
8/// A partial order is defined via [`CapabilityMode::is_subset_of`]:
9/// `ReadOnly < ReadWrite < All` and `ReadOnly < Execute < All`.
10/// `ReadWrite` and `Execute` are *incomparable* (neither is a subset
11/// of the other). `fork_session` enforces `child <= parent`.
12#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, serde::Serialize, serde::Deserialize)]
13#[serde(rename_all = "snake_case")]
14pub enum CapabilityMode {
15    /// Reading and searching only. No edits, no shell, no background tasks.
16    ReadOnly,
17    /// Read + edit. No shell execution.
18    ReadWrite,
19    /// Read + shell execution + background-task control. No edits.
20    Execute,
21    /// Every tool kind allowed.
22    All,
23}
24
25impl Default for CapabilityMode {
26    /// Defaults to [`CapabilityMode::ReadWrite`] (subagent default;
27    /// the main/root session is always `All`).
28    fn default() -> Self {
29        Self::ReadWrite
30    }
31}
32
33impl CapabilityMode {
34    /// Filter `config.tools` by capability mode, returning a copy with
35    /// disallowed tools dropped.
36    ///
37    /// Tools whose `kind` is `None` (baseline, e.g. ad-hoc tools
38    /// declared via `ToolConfig::simple`) are preserved across all
39    /// modes. **MCP-origin** `kind: None` tools are NOT preserved by
40    /// this method; see `resolve_session_toolset` for the asymmetric
41    /// handling.
42    pub fn filter(self, config: &ToolServerConfig) -> ToolServerConfig {
43        let kept: Vec<ToolConfig> = config
44            .tools
45            .iter()
46            .filter(|tool| match tool.kind {
47                None => true,
48                Some(kind) => kind_allowed(self, kind),
49            })
50            .cloned()
51            .collect();
52        ToolServerConfig {
53            tools: kept,
54            behavior_preset: config.behavior_preset.clone(),
55        }
56    }
57
58    /// Whether every kind allowed by `self` is also allowed by `other`.
59    /// Used by `fork_session` to reject capability widening.
60    pub fn is_subset_of(self, other: CapabilityMode) -> bool {
61        for kind in ALL_TOOL_KINDS {
62            if kind_allowed(self, *kind) && !kind_allowed(other, *kind) {
63                return false;
64            }
65        }
66        true
67    }
68}

116/// Maps `(CapabilityMode, ToolKind)` -> kept-or-dropped.
117///
118/// This `match` is intentionally exhaustive: when `ToolKind` gains a
119/// new variant the compiler errors here, forcing a triage decision.
120pub(crate) fn kind_allowed(mode: CapabilityMode, kind: ToolKind) -> bool {
121    use CapabilityMode as M;
122    use ToolKind::*;
123
124    if matches!(mode, M::All) {
125        return true;
126    }
127
128    match kind {
129        // Meta tools: always allowed.
130        Plan | EnterPlan | ExitPlan | AskUser | Skill | SearchTool | GoalUpdate => true,
131
132        // Read class.
133        Read | MemoryGet | MemorySearch => {
134            matches!(mode, M::ReadOnly | M::ReadWrite | M::Execute)
135        }
136
137        // Search class.
138        Search | WebSearch | WebFetch => {
139            matches!(mode, M::ReadOnly | M::ReadWrite | M::Execute)
140        }
141
142        // Inspect class.
143        Lsp | ListDir | List => matches!(mode, M::ReadOnly | M::ReadWrite | M::Execute),
144
145        // Edit class.
146        Edit | Write | Delete | Move | ImageGen | VideoGen | ImageToVideo | ReferenceToVideo
147        | DeployApp => matches!(mode, M::ReadWrite),
148
149        // Bash / shell.
150        Execute => matches!(mode, M::Execute),
151
152        // Process control (background tasks, monitors).
153        BackgroundTaskAction | WaitTasksAction | KillTaskAction | Task | Monitor => {
154            matches!(mode, M::Execute)
155        }
156
157        // Integration dispatch.
158        UseTool => matches!(mode, M::ReadWrite | M::Execute),
159
160        // Catch-all -- only `All` mode keeps it (early-return above).
161        Other => false,
162    }
163}

28/// contribute additional tool registrations into every
29/// [`ToolRegistryBuilder::new`].
30///
31/// This inverts the dependency for harness code that must live outside
32/// this crate: instead of `xai-grok-tools` referencing an out-of-tree tool
33/// pack, the pack calls
34/// [`register_tool_pack`] at startup and registers itself here.
35///
36/// # Ordering contract
37/// [`register_tool_pack`] MUST run before the FIRST `ToolRegistryBuilder::new()`
38/// in the process. Packs registered after a builder
39/// has been constructed do not retroactively apply to that builder.
40/// A tool pack: a function that contributes registrations to a builder.

47/// contract. Idempotency is the caller's responsibility (a pack registered
48/// twice registers its tools twice).

1385    /// implementations that never call `use_tool` themselves, so they do

1433    /// Dispatch a tool call by client-facing name with client-facing params.
1434    ///
1435    /// `cwd_override` — optional per-call working directory. When `Some`, tools
1436    /// will use this instead of the session `Cwd` from Resources. This is
1437    /// stack-local (not shared state), so concurrent calls with different
1438    /// overrides don't race.
1439    pub async fn call(
1440        self: &Arc<Self>,
1441        tool_name: &str,
1442        tool_args: serde_json::Value,
1443        tool_call_id: &str,
1444        cwd_override: Option<std::path::PathBuf>,
1445    ) -> Result<ToolRunResult, xai_tool_runtime::ToolError> {

1447        let mut stream = self.call_streaming(tool_name, tool_args, tool_call_id, cwd_override);
1448        while let Some(item) = stream.next().await {
1449            match item {
1450                xai_tool_runtime::ToolStreamItem::Progress(_) => continue,
1451                xai_tool_runtime::ToolStreamItem::Terminal(result) => return result,
1452            }
1453        }

1455    }
1456    /// Streaming sibling of [`call`].
1457    ///
1458    /// Forwards every inner [`ToolStreamItem::Progress`] unchanged, and when the
1459    /// inner dispatch stream reaches its terminal, runs the shared
1460    /// post-processing tail ([`finalize_output`]) on it and yields the resulting
1461    /// [`ToolRunResult`] as the single terminal of the outer stream.
1462    ///
1463    /// This is a non-`async` inherent method: it synchronously builds and
1464    /// returns a `'static` boxed stream. Because [`ToolStream`] is `'static`,
1465    /// all `.await` and `Arc::clone(self)` happen *inside* the stream block so
1466    /// nothing borrows `self` across the stream.
1467    ///
1468    /// [`ToolStream`]: xai_tool_runtime::ToolStream
1469    /// [`ToolStreamItem::Progress`]: xai_tool_runtime::ToolStreamItem::Progress
1470    pub fn call_streaming(
1471        self: &Arc<Self>,
1472        tool_name: &str,
1473        tool_args: serde_json::Value,
1474        tool_call_id: &str,
1475        cwd_override: Option<std::path::PathBuf>,
1476    ) -> xai_tool_runtime::ToolStream<ToolRunResult> {
1477        use futures::StreamExt;
1478        let this = Arc::clone(self);
1479        let tool_name = tool_name.to_owned();
1480        let tool_call_id = tool_call_id.to_owned();
1481        Box::pin(async_stream::stream! {
1482            let parts = match this.prepare_dispatch(& tool_name, tool_args, &
1483            tool_call_id, cwd_override,) { Ok(parts) => parts, Err(e) => { yield
1484            xai_tool_runtime::ToolStreamItem::Terminal(Err(e)); return; } }; let
1485            DispatchParts { lr_handle, ctx, canonical_params, output_converter,
1486            effective_tool_name, } = parts; let mut inner = lr_handle.execute(ctx,
1487            canonical_params). await; while let Some(item) = inner.next(). await {
1488            match item { xai_tool_runtime::ToolStreamItem::Progress(p) => { yield
1489            xai_tool_runtime::ToolStreamItem::Progress(p); }
1490            xai_tool_runtime::ToolStreamItem::Terminal(Err(e)) => { yield
1491            xai_tool_runtime::ToolStreamItem::Terminal(Err(e)); return; }
1492            xai_tool_runtime::ToolStreamItem::Terminal(Ok(typed)) => { let run_result
1493            = this.finalize_output(typed.value, & output_converter,
1494            effective_tool_name). await; yield
1495            xai_tool_runtime::ToolStreamItem::Terminal(run_result); return; } } }
1496            yield
1497            xai_tool_runtime::ToolStreamItem::Terminal(Err(stream_no_terminal_error()));
1498        })
1499    }
