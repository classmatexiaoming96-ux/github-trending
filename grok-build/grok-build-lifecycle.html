149    /// Coarse activity of a resident session for the dashboard status column.
150    ///
151    /// Precedence: a non-empty pending-interaction map →
152    /// `NeedsInput` (wins even over a running turn — a session awaiting a
153    /// permission *mid-turn* is "needs input"); else a running turn →
154    /// `Working`; else map the coarse `SessionLiveState`.
155    pub(super) fn resident_activity(
156        &self,
157        id: &acp::SessionId,
158    ) -> crate::agent::roster::RosterActivity {
159        use crate::agent::roster::RosterActivity;
160        let (needs_input, turn_running) = self
161            .sessions
162            .borrow()
163            .get(id)
164            .map(|h| {
165                let needs_input = h
166                    .pending_interactions
167                    .lock()
168                    .map(|g| !g.is_empty())
169                    .unwrap_or(false);
170                let turn_running = h
171                    .current_prompt_id
172                    .lock()
173                    .map(|g| g.is_some())
174                    .unwrap_or(false);
175                (needs_input, turn_running)
176            })
177            .unwrap_or((false, false));
178        if needs_input {
179            return RosterActivity::NeedsInput;
180        }
181        if turn_running {
182            return RosterActivity::Working;
183        }
184        match self.session_live_state.borrow().get(id).copied() {
185            Some(SessionLiveState::Completed) => RosterActivity::Completed,
186            Some(SessionLiveState::DeadFailed) => RosterActivity::Dead,
187            Some(SessionLiveState::Dormant) => RosterActivity::Dormant,
188            _ => RosterActivity::Idle,
189        }
190    }

111/// one WARN per GET (set RUST_LOG=rmcp=warn to see them).
112#[tokio::test(flavor = "multi_thread")]
113async fn repro_zero_backoff_reconnect_flood() {
114    let (url, gets) = spawn_fake_server(GetBehavior::AbnormalBodyDeath).await;
115    let transport = StreamableHttpClientTransport::from_uri(url.as_str());
116    let client = ().serve(transport).await.expect("handshake against fake server should succeed");
117
118    const OBSERVE: Duration = Duration::from_secs(3);
119    tokio::time::sleep(OBSERVE).await;
120    let n = gets.load(Ordering::Relaxed);
121    let _ = client.cancel().await;
122
123    eprintln!(
124        "[repro] {n} standing-GET reconnect attempts (= WARN log lines) in {OBSERVE:?} \
125         = {:.0}/sec",
126        n as f64 / OBSERVE.as_secs_f64()
127    );
128    // A backoff-respecting client would attempt ~3-5 in 3s; the bug produces
129    // hundreds-to-thousands.
130    assert!(
131        n > 20,
132        "expected a zero-backoff reconnect flood (>20 GETs in {OBSERVE:?}), got {n}; \
133         if this FAILS with a small n, the rmcp loop got fixed - delete mcp_http_client.rs"
134    );
135}

137/// The fix: same body-killing server, client wrapped in `McpHttpClient` —
138/// the backoff schedule allows only a handful of reconnects.
139#[tokio::test(flavor = "multi_thread")]
140async fn throttled_client_bounds_the_flood() {
141    let (url, gets) = spawn_fake_server(GetBehavior::AbnormalBodyDeath).await;
142    let throttled = McpHttpClient::new(
143        reqwest::Client::default(),
144        "fake-server",
145        WarnBudget::default(),
146    );
147    let transport = StreamableHttpClientTransport::with_client(
148        throttled,
149        StreamableHttpClientTransportConfig::with_uri(url.as_str()),
150    );
151    let client = ().serve(transport).await.expect("handshake against fake server should succeed");
152
153    // Checkpoint: backoff must engage early (instant, instant, 0.5s => 3-4
154    // GETs by 1.2s; an unthrottled client would be in the hundreds).
155    tokio::time::sleep(Duration::from_millis(1200)).await;
156    let early = gets.load(Ordering::Relaxed);
157    assert!(
158        early <= 4,
159        "backoff must engage early; got {early} GETs by 1.2s"
160    );
161
162    const OBSERVE: Duration = Duration::from_secs(4);
163    tokio::time::sleep(OBSERVE - Duration::from_millis(1200)).await;
164    let n = gets.load(Ordering::Relaxed);
165    let _ = client.cancel().await;
166
167    eprintln!("[repro] {n} throttled reconnect attempts in {OBSERVE:?} (schedule allows ~5-6)");
168    assert!(
169        (3..=8).contains(&n),
170        "throttled reconnects should follow the backoff schedule (~5-6 in {OBSERVE:?}), got {n}"
