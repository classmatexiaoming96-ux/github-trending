103/**
104 * In-memory registry of connectors. A single shared instance lives in
105 * `./index.ts`; feature agents call `.register()` there.
106 */
107export class ConnectorRegistry {
108  private readonly connectors = new Map<string, Connector>()
109
110  register(connector: Connector): void {
111    if (this.connectors.has(connector.id)) {
112      throw new Error(`Connector "${connector.id}" is already registered`)
113    }
114    this.connectors.set(connector.id, connector)
115  }
116
117  get(id: string): Connector | undefined {
118    return this.connectors.get(id)
119  }
120
121  has(id: string): boolean {
122    return this.connectors.has(id)
123  }
124
125  all(): Connector[] {
126    return [...this.connectors.values()]
127  }
128
129  byDomain(domain: ConnectorDomain): Connector[] {
130    return this.all().filter((c) => c.domain === domain)
131  }
132
133  /** Serializable catalog for tools / UI (drops the search/fetch functions). */
134  catalog(): CatalogEntry[] {
135    return this.all().map(({ id, name, domain, description, homepage }) => ({
136      id,
137      name,
138      domain,
139      description,
140      homepage,
141    }))
142  }
§ 02 · KEY FLOW

关键流程

connector impl → Connector contract → registry.register → science_search / science_list_dbs → workspace artifact

这条路径把外部数据库差异藏在连接器内部。backend/cli/src/science/connectors/types.ts:26 的 ConnectorHit 统一 id/title/summary/url/score，backend/cli/src/science/connectors/types.ts:62 的 RateLimit 给 HTTP 层限流提示，backend/cli/src/science/connectors/index.ts:35 再导出 types 与 http，让实现层共享同一工具箱。

▸为什么这样拆

这个子模块的拆法不是按文件大小，而是按运行时责任边界来切。主页面已经把 backend/cli/src/index.ts:55、backend/cli/src/config/config.ts:75、backend/cli/src/provider/provider.ts:57、backend/cli/src/science/connectors/types.ts:76 和 tooling/plugin/src/index.ts:148 串成整体；子页面只展开其中一个闭环，避免把所有源码证据挤成行号清单。

§ 03 · EDGE CASES

边界 / 失败模式

  • Registry 只防重复 id，见 backend/cli/src/science/connectors/types.ts:111；连接器结果质量仍依赖具体实现。
  • RateLimit 是连接器可请求的策略，backend/cli/src/science/connectors/types.ts:69 到 backend/cli/src/science/connectors/types.ts:73 没有强制每个实现都声明。
  • Catalog 在 backend/cli/src/science/connectors/types.ts:133 会丢弃函数，适合 UI，但调试时要回到具体 connector 源码。
