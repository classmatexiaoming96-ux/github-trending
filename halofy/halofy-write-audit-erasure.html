  24 * Deterministic JSON of an arbitrary JSON value with object keys sorted at
  25 * every depth. Arrays keep their order (erasedIds ordering is meaningful and
  26 * fixed by the resolver); objects (including the nested ForgetTarget) sort.
  27 */
  28function stableStringify(value: unknown): string {
  29  if (value === null || typeof value !== "object") {
  30    return JSON.stringify(value);
  31  }
  32  if (Array.isArray(value)) {
  33    return `[${value.map((v) => stableStringify(v)).join(",")}]`;
  34  }
  35  const obj = value as Record<string, unknown>;
  36  const keys = Object.keys(obj).sort();
  37  const parts = keys.map(
  38    (k) => `${JSON.stringify(k)}:${stableStringify(obj[k])}`,
  39  );
  40  return `{${parts.join(",")}}`;
  41}

  68 * Verify a certificate against an out-of-band trusted public key. The inline
  69 * key is signed descriptive metadata and must identify that same trusted key;
  70 * it is never itself a trust anchor. Rebuilds the canonical payload (dropping
  71 * `signature`) and checks the ed25519 signature.
  72 * Returns false — never throws — for a key mismatch, tampered field, bad
  73 * signature, malformed key, or non-ed25519 algorithm.
  74 */
  75export function verifyCertificate(
  76  cert: DeletionCertificate,
  77  trustedPublicKeyBase64: string,
  78): boolean {
  79  try {
  80    if (cert.algorithm !== "ed25519") return false;
  81    const inlineKeyDer = Buffer.from(cert.publicKey, "base64");
  82    const trustedKeyDer = Buffer.from(trustedPublicKeyBase64, "base64");
  83    if (
  84      inlineKeyDer.length === 0 ||
  85      inlineKeyDer.length !== trustedKeyDer.length ||
  86      !timingSafeEqual(inlineKeyDer, trustedKeyDer)
  87    ) {
  88      return false;
  89    }
  90    const publicKey = createPublicKey({
  91      key: trustedKeyDer,
  92      format: "der",
  93      type: "spki",
  94    });

  96    const bytes = Buffer.from(canonicalPayload(cert), "utf8");
  97    const signature = Buffer.from(cert.signature, "base64");
  98    // ed25519: algorithm is null (the scheme fixes the digest).
  99    return cryptoVerify(null, bytes, publicKey, signature);
 100  } catch {

 102  }
