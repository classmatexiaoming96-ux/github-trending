32    let ws: WebSocket | null = null;
33    let alive = true;
34    let retries = 0;
35    let everConnected = false;
36    let queue: Promise<void> = Promise.resolve();

38    const connect = () => {
39      if (!alive) return;
40      const url = new URL('/api/studio/bridge', location.href);
41      url.protocol = url.protocol === 'https:' ? 'wss:' : 'ws:';
42      let sock: WebSocket;
43      try {
44        sock = new WebSocket(url);
45      } catch {
46        return; // environment unsupported, give up
47      }
48      ws = sock;
49      sock.onopen = () => {
50        everConnected = true;
51        retries = 0;
52      };
53      sock.onmessage = (ev) => {
54        if (typeof ev.data !== 'string' || ev.data === 'pong') return;
55        let m: { id?: string; tool?: string; input?: Record<string, unknown> };
56        try {
57          m = JSON.parse(ev.data) as typeof m;
58        } catch {
59          return;
60        }
61        if (!m.id || !m.tool) return;
62        const { id, tool, input } = m as { id: string; tool: string; input?: Record<string, unknown> };
63        queue = queue.then(async () => {
64          let out: StudioToolResult & { state?: string };
65          try {
66            out = tool === 'get_state' ? { ok: true, state: optsRef.current.getState() } : await optsRef.current.runTool(tool, input ?? {});
67          } catch (e) {
68            out = { ok: false, error: e instanceof Error ? e.message : String(e) };
69          }
70          try {
71            sock.send(JSON.stringify({ id, ...out }));
72          } catch {
73            /* socket already dead, DO side will time out / fail */
74          }
75          optsRef.current.onExternalCall?.(tool, out);
76        });
77      };
78      sock.onclose = (ev) => {
79        if (ws === sock) ws = null;
80        if (!alive) return;
81        if (ev.code === 4000) return; // kicked by a new tab: this page exits the bridge, don't fight for it
82        if (!everConnected && retries >= 6) return; // never connected (not logged in, etc.): don't spin
83        setTimeout(connect, Math.min(30_000, 1_000 * 2 ** retries++));
84      };
