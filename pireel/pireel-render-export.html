32export async function probeVideoFile(file: File): Promise<ProbedFile> {
33  const { ALL_FORMATS, BlobSource, Input } = await import('mediabunny');
34  const input = new Input({ source: new BlobSource(file), formats: ALL_FORMATS });
35  try {
36    const v = await input.getPrimaryVideoTrack();
37    const a = await input.getPrimaryAudioTrack().catch(() => null);
38    // computeDuration measures the "max end timestamp": an mp4 with a non-zero first packet inflates the
39    // duration by that offset. Subtract getFirstTimestamp to convert to <video> playback frame (currentTime 0 = earliest sample).
40    const duration = (await input.computeDuration()) - Math.max(0, await input.getFirstTimestamp());

42      durationSec: Number.isFinite(duration) && duration > 0 ? duration : 0,
43      width: v?.displayWidth ?? 0,
44      height: v?.displayHeight ?? 0,
45      hasAudio: !!a,
46    };
47  } finally {

60  const r = await fetch('/api/auto-edit/asr', {
61    method: 'POST',
62    headers: { 'content-type': 'application/json' },
63    body: JSON.stringify({ audio_url: url }),
64  });
65  // Server failures must throw a clear error, not silently become an empty array — callers need to tell "ASR failed" apart from "the video truly has no speech"
66  if (!r.ok) throw new Error(t('ASR 请求失败(HTTP {status})', { status: r.status }));
67  const j = (await r.json()) as { segments?: Array<{ start: number; end: number; text: string }> };
68  // ASR times are in the "audio track's own zero" frame (audio extraction subtracts the audio track's first
69  // packet), while playback is in "earliest sample across all tracks". Files whose two tracks' first packets
70  // are out of sync need this delta added back, or captions/cut points shift wholesale.
71  const off = await audioPlaybackOffset(file);
72  const segs: AsrSegment[] = (Array.isArray(j.segments) ? j.segments : [])
73    .filter((s) => s.text?.trim())
74    .map((s) => ({ start: Math.max(0, s.start + off), end: Math.max(s.start + off + 0.1, s.end + off), text: s.text.trim() }));
75  if (segs.length) setCachedAsr(sig, segs);
76  return segs;
77}

  • 浏览器缺 WebCodecs 时直接报错，不再生成未经验证的黑屏服务端输出。
  • foreignObject 不能拉外部图片，所以导出前必须把图片内联。
  • 长项目如果序列化全部 DOM，会让每一帧承担全部 block 成本，因此只取可见 children。
