62/**
63 * Resolve diff from/to/paths with git-diff defaults, treating on-disk path
64 * positionals as path filters when they are not valid git refs.
65 *
66 * Examples:
67 * - `diff main src` → main vs worktree, paths=[src]
68 * - `diff main feature src` → main vs feature, paths=[src]
69 * - `diff src` → HEAD vs worktree, paths=[src]
70 */
71export function resolveDiffSnapshotsAndPaths(
72  cwd: string,
73  from: string | undefined,
74  to: string | undefined,
75  paths: string[],
76): { from: Snapshot; to: Snapshot; paths: string[] } {
77  if (from === undefined && to === undefined) {
78    return { ...resolveSnapshots(undefined, undefined), paths };
79  }
80

82    if (isCommitRef(cwd, from)) {
83      return { ...resolveSnapshots(from, undefined), paths };
84    }
85    if (isPathOnDisk(cwd, from)) {
86      return { ...resolveSnapshots(undefined, undefined), paths: [from, ...paths] };
87    }
88    throw new Error(`Unknown git ref: ${from}`);
89  }
90
91  if (from !== undefined && to !== undefined) {
92    if (!isCommitRef(cwd, from)) {
93      throw new Error(`Unknown git ref: ${from}`);
94    }
95    if (isCommitRef(cwd, to)) {
96      return { ...resolveSnapshots(from, to), paths };
97    }
98    if (isPathOnDisk(cwd, to)) {
99      return {
100        ...resolveSnapshots(from, undefined),
101        paths: [to, ...paths],
102      };
103    }
104    throw new Error(`Unknown git ref: ${to}`);
105  }
106
107  // to without from shouldn't happen via CLI positionals, but honor options.
108  return { ...resolveSnapshots(from, to), paths };

278function chunkBlobs(files: SnapshotFile[]): Blob[][] {
279  const unique = new Map<string, number>();
280  for (const file of files) {
281    if (file.oid && file.size !== undefined && !unique.has(file.oid)) {
282      unique.set(file.oid, file.size);
283    }
284  }
285
286  const chunks: Blob[][] = [];
287  let chunk: Blob[] = [];
288  let bytes = 0;
289  for (const [oid, size] of unique) {
290    if (chunk.length > 0 && bytes + size > BATCH_BYTES) {
291      chunks.push(chunk);
292      chunk = [];
293      bytes = 0;
294    }
295    chunk.push({ oid, size });
296    bytes += size;
297  }
298  if (chunk.length > 0) chunks.push(chunk);

302function readBlobBatch(cwd: string, batch: Blob[]): Map<string, string> {
303  const input = Buffer.from(`${batch.map((blob) => blob.oid).join("\n")}\n`);
304  const bytes = batch.reduce((total, blob) => total + blob.size, 0);
305  const maxBuffer = bytes + batch.length * 128 + 1024;
306  const output = gitBuffer(cwd, ["cat-file", "--batch"], input, maxBuffer);
307  const blobs = new Map<string, string>();
308  let offset = 0;
309
310  while (offset < output.length) {
311    const headerEnd = output.indexOf(0x0a, offset);
312    if (headerEnd < 0) throw new Error("Invalid git cat-file response");
313
314    const header = output.subarray(offset, headerEnd).toString("ascii");
315    const [oid, type, sizeText] = header.split(" ");
316    const size = Number(sizeText);
317    if (!oid || type !== "blob" || !Number.isInteger(size)) {
318      throw new Error(`Invalid git cat-file header: ${header}`);
319    }
320
321    const contentStart = headerEnd + 1;
322    const contentEnd = contentStart + size;
323    if (contentEnd >= output.length) {
324      throw new Error(`Truncated git blob: ${oid}`);
325    }
326
327    blobs.set(oid, output.subarray(contentStart, contentEnd).toString("utf8"));
328    offset = contentEnd + 1;
329  }
330
331  return blobs;
332}

▸ASCII fast path 与 structured format

73/**
74 * Default: classic ASCII (TTY + pipes) with optional CTAs.
75 * `--format` / `--json`: structured envelope for agents.
76 *
77 * The ASCII path writes to stdout itself so that piped output stays raw text
78 * (incur wraps a scalar payload as `{ data, cta }` once a CTA is attached).
79 * `--token-count` / `--token-limit` / `--token-offset` act on what a command
80 * *returns*, though, so under those flags we hand the ASCII back to incur and
81 * let it do the counting and slicing. The CTA is dropped there on purpose: it
82 * keeps the counted output exactly the tree the user asked about.
83 */
84function emitAsciiOrData(
85  c: EmitContext,
86  result: DiffResult | TreeResult | ReachResult,
87  cta?: CtaMeta,
88): unknown {
89  if (!c.formatExplicit) {
90    if (tokenFlagActive) return result.ascii;
91    process.stdout.write(
92      result.ascii.endsWith("\n") ? result.ascii : `${result.ascii}\n`,
93    );
94    if (cta) return c.ok({}, cta);
95    return;
96  }
