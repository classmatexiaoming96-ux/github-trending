packages/core/src/app-keys/injector.ts · L5-50TypeScript
5/**
6 * Owns the stored-key slice of `process.env`. Two invariants:
7 *
8 * - The real process environment always wins. A name that already had a value
9 *   before injection is never overwritten — operator config (env files, CI,
10 *   container env) cannot be shadowed from the dashboard. Such keys are
11 *   reported as overridden instead of silently going live.
12 * - Only names this injector set are ever deleted or scrubbed. It tracks its
13 *   own writes, so removing a stored key can never unset an operator-set var
14 *   that happens to share the name.
15 */
16export class AppKeyInjector {
17  private injected = new Set<string>();
18  private overridden = new Set<string>();
19
20  constructor(private env: NodeJS.ProcessEnv = process.env) {}
21
22  /** Applies a stored key. Returns true when the value went live, false when a
23   * pre-existing environment value kept precedence. */
24  apply(name: string, value: string): boolean {
25    if (!this.injected.has(name) && this.env[name] !== undefined) {
26      this.overridden.add(name);
27      logger.warn(
28        `App key ${name} is overridden by the process environment; the stored value is not live`,
29      );
30      return false;
31    }
32    this.env[name] = value;
33    this.injected.add(name);
34    return true;
35  }
36
37  /** Removes a stored key from the environment, only if this injector set it.
38   * Returns true when the environment actually changed. */
39  remove(name: string): boolean {
40    this.overridden.delete(name);
41    if (!this.injected.has(name)) return false;
42    delete this.env[name];
43    this.injected.delete(name);
44    return true;
45  }
46
47  isOverridden(name: string): boolean {
48    return this.overridden.has(name);
49  }
50}
