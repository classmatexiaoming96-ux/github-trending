9// Node's ESM resolver drops the query string when a module resolves its own
10// static imports, so the ?v= cache-buster on an entry URL re-evaluates only
11// the entry file: a helper it imports keeps its unsalted URL and stays cached
12// with whatever it captured at module scope. These hooks give the entry's
13// whole own-file graph one identity per (file stats, env epoch):
14//
15// - resolve: short-circuits already-salted file URLs before any other loader
16//   sees them (tsx — the `source` container mode — otherwise normalizes the
17//   query away, collapsing every epoch onto one cached instance), and
18//   propagates the parent's salt onto relative imports.
19// - load: serves salted URLs from disk itself, because downstream loaders
20//   fail on file URLs that carry a query.
21//
22// Only compiled `.js`/`.mjs` files are salted: TypeScript app sources need
23// tsx's transform pipeline, which the bypass would skip — source-mode
24// workspace apps keep their existing repack-or-restart reload story. Bare
25// specifiers stay untouched on purpose: node_modules must resolve to the one
26// shared instance per process.
27const SALT_QUERY = /\?v=[^&]*$/;

30  const path = url.split("?")[0];

34registerHooks({

51    return result;
52  },

101export async function resolveModuleEntryPath(baseDir: string, entry?: string): Promise<string> {
102  const absoluteBaseDir = resolve(baseDir);

113  ]);
