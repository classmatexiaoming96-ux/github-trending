113def extract_created_event_id(text: str) -> str | None:
114    """Strict single-created-event shape check shared by every wire adapter.
115
116    Accepts exactly the creation-tool payload shape: a JSON dict with a
117    top-level ``"event"`` dict whose ``event_id`` is a well-formed evt id.
118    Rejects (belt-and-braces — the tool-name allowlist already excludes read
119    tools): list-shaped payloads carrying ``"events"``/``"runs"`` lists,
120    error texts, truncated/torn JSON, outcome-only machine_check payloads,
121    and any non-dict shape.
122    """
123
124    try:
125        payload = json.loads(text)
126    except (json.JSONDecodeError, ValueError):
127        return None
128    if not isinstance(payload, dict):
129        return None
130    event = payload.get("event")
131    if not isinstance(event, dict):
132        return None
133    if any(isinstance(payload.get(key), list) for key in ("events", "runs")):
134        return None
135    event_id = event.get("event_id")
136    if not isinstance(event_id, str) or not EVIDENCED_EVENT_ID_RE.match(event_id):
137        return None
138    return event_id
