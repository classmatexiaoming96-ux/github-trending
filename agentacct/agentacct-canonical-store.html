132@dataclass(frozen=True)
133class SchemaMigration:
134    """One forward schema step from ``from_version`` to ``from_version + 1``.
135
136    Statements run one by one inside a single transaction (never via
137    ``executescript``, which would commit early) and must produce exactly the
138    DDL that compiling the target build's packaged ``schema.sql`` produces —
139    the open-time inventory comparison enforces that equivalence.
140    """
141
142    from_version: int
143    name: str
144    statements: tuple[str, ...]
145
146    @property
147    def checksum(self) -> str:
148        digest = hashlib.sha256(b"agent-chronicle-schema-migration-v1\x00")
149        for statement in self.statements:
150            payload = statement.encode("utf-8")
151            digest.update(len(payload).to_bytes(8, "big"))
152            digest.update(payload)
153        return digest.hexdigest()
154

228def _apply_schema_migrations(
229    connection: sqlite3.Connection,
230    *,
231    from_version: int,
232    app_version: str,
233) -> int:
234    journal_mode = str(connection.execute("PRAGMA journal_mode").fetchone()[0]).lower()
235    if journal_mode != "wal":
236        raise ValueError("candidate database is not in WAL mode; refusing to migrate")
237    metadata_version = connection.execute(
238        "SELECT schema_version FROM store_metadata WHERE singleton = 1"
239    ).fetchone()
240    if metadata_version is None or int(metadata_version[0]) < from_version:
241        raise ValueError("candidate metadata disagrees with the schema version; refusing to migrate")
242    current = from_version
243    while current < SCHEMA_VERSION:
244        step = SCHEMA_MIGRATIONS.get(current)
245        if step is None:
246            raise ValueError(
247                f"candidate schema version {current} has no upgrade path in "
248                f"this build; rebuild the candidate from a fresh snapshot"
249            )
250        if step.from_version != current:
251            raise ValueError(
252                f"schema migration '{step.name}' is registered for version "
253                f"{current} but declares from_version {step.from_version}"
254            )
255        target = current + 1
256        try:
257            connection.execute("BEGIN IMMEDIATE")
258        except sqlite3.OperationalError as exc:
259            raise ValueError(
260                f"schema migration '{step.name}' to version {target} could not "
261                f"acquire the write lock; another process may be migrating"
262            ) from exc
263        try:
264            # Re-read versions under the exclusive lock: a concurrent writable
265            # open may have finished this step (or the whole chain) while we
266            # waited. Adopting its committed progress keeps racing opens
267            # idempotent instead of failing on the stamping UNIQUE constraint.
268            observed = int(connection.execute("PRAGMA user_version").fetchone()[0])
269            row = connection.execute(
270                "SELECT schema_version FROM store_metadata WHERE singleton = 1"
271            ).fetchone()
272            stamped = None if row is None else int(row[0])
273            if observed != current or stamped != current:
274                if stamped == observed and current < observed <= SCHEMA_VERSION:
275                    connection.execute("ROLLBACK")
276                    current = observed
277                    continue
278                raise ValueError(
279                    "candidate metadata disagrees with the schema version; refusing to migrate"
280                )
281            for statement in step.statements:
282                connection.execute(statement)
283                if not connection.in_transaction:
284                    # A statement that is itself COMMIT/ROLLBACK/END would
285                    # make the stamping writes below autocommit one by one,
286                    # silently breaking per-step atomicity.
287                    raise ValueError(
288                        f"schema migration '{step.name}' statement ended the "
289                        f"transaction; step atomicity cannot be proved"
290                    )
291            connection.execute(
292                "INSERT INTO schema_migrations(version, name, checksum, app_version, applied_at_us) "
293                "VALUES (?, ?, ?, ?, ?)",
294                (target, step.name, step.checksum, app_version, now_us()),
295            )
296            connection.execute(
297                "UPDATE store_metadata SET schema_version = ? WHERE singleton = 1",
298                (target,),
299            )
300            connection.execute(f"PRAGMA user_version = {target}")
301            connection.execute("COMMIT")
302        except BaseException as exc:
303            if connection.in_transaction:
304                try:
305                    connection.execute("ROLLBACK")
306                except sqlite3.Error:
307                    pass
308            if isinstance(exc, sqlite3.Error):
309                # A distinct error class keeps migration failures out of the
310                # generic unreadable-DDL diagnosis in open().
311                raise ValueError(
312                    f"schema migration '{step.name}' to version {target} failed"
313                ) from exc
314            raise
315        current = target
316    return current
§ 02 · KEY FLOW

关键流程

candidate path ─▶ schema checksum ─▶ WAL transaction ─▶ migration stamp ─▶ read model health

▸实现判断

它不像普通 sqlite helper 那样自动选默认路径；候选库必须显式创建，迁移也要持有写锁、验证 metadata、逐条执行 DDL 并 stamp schema_migrations。 这条路径还可以和主页面的 src/agent_chronicle/source_discovery.py:37、src/agent_chronicle/capture_runtime.py:41、src/agent_chronicle/control_web.py:50 连起来看：agentacct 的策略是收集证据但不收集敏感正文，展示状态但不泄露执行权柄。

§ 03 · EDGE CASES

边界 / 失败模式

  • src/agent_chronicle/canonical/sqlite.py:50 附近的入口如果收到不可信输入，只能产出降级或拒绝，不得写入 raw prompt/response。
  • src/agent_chronicle/canonical/sqlite.py:132 的分支必须和测试夹具保持同步，避免客户端格式漂移后静默误归因。
